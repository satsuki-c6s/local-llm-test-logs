```html
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>ブロック崩し</title>
<style>
  body {
    margin: 0;
    background: #1a1a2e;
    display: flex;
    justify-content: center;
    align-items: center;
    height: 100vh;
  }
  canvas {
    background: #16213e;
    border: 2px solid #4a4e69;
    border-radius: 8px;
  }
</style>
</head>
<body>
<canvas id="game" width="600" height="500"></canvas>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');

// 定数
const W = canvas.width;
const H = canvas.height;
const PADDLE_W = 80;
const PADDLE_H = 12;
const PADDLE_Y = H - 30;
const BALL_R = 8;
const BLOCK_COLS = 8;
const BLOCK_ROWS = 5;
const BLOCK_W = 60;
const BLOCK_H = 20;
const BLOCK_GAP = 10;
const BLOCK_TOP = 60;
const BLOCK_LEFT = (W - (BLOCK_COLS * BLOCK_W + (BLOCK_COLS - 1) * BLOCK_GAP)) / 2;

// 状態
let paddleX = W / 2 - PADDLE_W / 2;
let ballX, ballY, ballVX, ballVY;
let blocks = [];
let score = 0;
let state = 'ready'; // ready / playing / gameover / clear
let keys = {};

function initBlocks() {
  blocks = [];
  const colors = ['#e63946', '#f4a261', '#f1c40d', '#2a9d8c', '#48bfe3'];
  for (let r = 0; r < BLOCK_ROWS; r++) {
    for (let c = 0; c < BLOCK_COLS; c++) {
      blocks.push({
        x: BLOCK_LEFT + c * (BLOCK_W + BLOCK_GAP),
        y: BLOCK_TOP + r * (BLOCK_H + BLOCK_GAP),
        color: colors[r],
        alive: true
      });
    }
  }
}

function resetBall() {
  ballX = W / 2;
  ballY = PADDLE_Y - BALL_R - 2;
  ballVX = 4;
  ballVY = -4;
}

function resetGame() {
  initBlocks();
  resetBall();
  score = 0;
  state = 'ready';
}

// 入力
document.addEventListener('keydown', e => {
  if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
    keys[e.key] = true;
    e.preventDefault();
  }
  if (e.key === ' ' && state === 'ready') {
    state = 'playing';
    e.preventDefault();
  }
  if (e.key === 'Enter' && (state === 'gameover' || state === 'clear')) {
    resetGame();
  }
});
document.addEventListener('keyup', e => {
  if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
    keys[e.key] = false;
  }
});

function update() {
  // パドル移動
  const speed = 7;
  if (keys['ArrowLeft']) paddleX -= speed;
  if (keys['ArrowRight']) paddleX += speed;
  paddleX = Math.max(0, Math.min(W - PADDLE_W, paddleX));

  if (state !== 'playing') return;

  // ボール移動
  ballX += ballVX;
  ballY += ballVY;

  // 壁と天井
  if (ballX - BALL_R <= 0) {
    ballX = BALL_R;
    ballVX = Math.abs(ballVX);
  }
  if (ballX + BALL_R >= W) {
    ballX = W - BALL_R;
    ballVX = -Math.abs(ballVX);
  }
  if (ballY - BALL_R <= 0) {
    ballY = BALL_R;
    ballVY = Math.abs(ballVY);
  }

  // パドル当たり
  if (ballVY > 0 &&
      ballY + BALL_R >= PADDLE_Y &&
      ballY - BALL_R <= PADDLE_Y + PADDLE_H &&
      ballX >= paddleX && ballX <= paddleX + PADDLE_W) {
    ballY = PADDLE_Y - BALL_R;
    // パドルの当たり位置で角度を変える
    const hitPos = (ballX - paddleX) / PADDLE_W; // 0~1
    const angle = (hitPos - 0.5) * Math.PI * 0.6; // -54°~+54°
    const speed = Math.hypot(ballVX, ballVY);
    ballVX = speed * Math.sin(angle);
    ballVY = -Math.abs(speed * Math.cos(angle));
  }

  // ブロック当たり
  for (const b of blocks) {
    if (!b.alive) continue;
    if (ballX + BALL_R > b.x && ballX - BALL_R < b.x + BLOCK_W &&
        ballY + BALL_R > b.y && ballY - BALL_R < b.y + BLOCK_H) {
      b.alive = false;
      score += 10;
      // 反発方向の判定
      const overlapLeft = ballX + BALL_R - b.x;
      const overlapRight = b.x + BLOCK_W - (ballX - BALL_R);
      const overlapTop = ballY + BALL_R - b.y;
      const overlapBottom = b.y + BLOCK_H - (ballY - BALL_R);
      const minOverlap = Math.min(overlapLeft, overlapRight, overlapTop, overlapBottom);
      if (minOverlap === overlapLeft || minOverlap === overlapRight) {
        ballVX = -ballVX;
      } else {
        ballVY = -ballVY;
      }
      break;
    }
  }

  // 落下
  if (ballY - BALL_R > H) {
    state = 'gameover';
  }

  // クリア判定
  if (blocks.every(b => !b.alive)) {
    state = 'clear';
  }
}

function draw() {
  ctx.clearRect(0, 0, W, H);

  // ブロック
  for (const b of blocks) {
    if (!b.alive) continue;
    ctx.fillStyle = b.color;
    ctx.fillRect(b.x, b.y, BLOCK_W, BLOCK_H);
  }

  // パドル
  ctx.fillStyle = '#ffffff';
  ctx.fillRect(paddleX, PADDLE_Y, PADDLE_W, PADDLE_H);

  // ボール
  ctx.beginPath();
  ctx.arc(ballX, ballY, BALL_R, 0, Math.PI * 2);
  ctx.fillStyle = '#ffd166';
  ctx.fill();

  // スコア
  ctx.fillStyle = '#ffffff';
  ctx.font = '16px sans-serif';
  ctx.textAlign = 'left';
  ctx.fillText('スコア: ' + score, 10, 25);

  // メッセージ
  ctx.textAlign = 'center';
  if (state === 'ready') {
    ctx.font = '24px sans-serif';
    ctx.fillText('スペースキーでスタート', W / 2, H / 2);
  } else if (state === 'gameover') {
    ctx.font = '32px sans-serif';
    ctx.fillStyle = '#e63946';
    ctx.fillText('ゲームオーバー', W / 2, H / 2 - 20);
    ctx.font = '16px sans-serif';
    ctx.fillStyle = '#ffffff';
    ctx.fillText('スコア: ' + score, W / 2, H / 2 + 10);
    ctx.fillText('Enterキーでリスタート', W / 2, H / 2 + 35);
  } else if (state === 'clear') {
    ctx.font = '32px sans-serif';
    ctx.fillStyle = '#2a9d8c';
    ctx.fillText('クリア!', W / 2, H / 2 - 20);
    ctx.font = '16px sans-serif';
    ctx.fillStyle = '#ffffff';
    ctx.fillText('スコア: ' + score, W / 2, H / 2 + 10);
    ctx.fillText('Enterキーでリスタート', W / 2, H / 2 + 35);
  }
}

function loop() {
  update();
  draw();
  requestAnimationFrame(loop);
}

resetGame();
loop();
</script>
</body>
</html>
```

**操作方法:**
- **スペースキー**: ゲーム開始
- **← / → 矢印キー**: パドル移動
- **Enterキー**: ゲームオーバー/クリア後のリスタート

- 5段 × 8列のブロック（10点ずつ、最大400点）
- パドルの当たり位置でボールの跳ね返り角度が変わる（左右の端で角度がきつくなる）
- 全ブロック消滅で「クリア!」、ボール落下で「ゲームオーバー」を表示
