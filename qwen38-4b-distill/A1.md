import csv
import os
import sys
from typing import Any


def convert_csv(
    input_path: str,
    output_path: str,
    encoding: str = "utf-8",
    errors: str = "replace"
) -> None:
    """
    Shift-JIS の CSV ファイルを読み込み、UTF-8 で保存し直す関数。

    Args:
        input_path: 入力ファイルパス（Shift-JIS 形式）
        output_path: 出力ファイルパス（UTF-8 形式）
        encoding: 入力ファイルのエンコーディング（デフォルト: Shift-JIS）
        errors: デコード失敗時のエラーハンドリング方法
               - 'replace': 不明な文字を ? に置き換える
               - 'ignore': 不明な文字を無視する
               - 'strict': エラーを発生させる（デフォルト）

    Raises:
        FileNotFoundError: 入力ファイルが存在しない場合
        UnicodeDecodeError: デコード失敗の場合
    """
    # 入力ファイルが存在するか確認
    if not os.path.exists(input_path):
        raise FileNotFoundError(
            f"入力ファイル '{input_path}' が見つかりません。"
        )

    # 入力ファイルのエンコーディングを確認（Shift-JIS の場合）
    input_encoding = "shift_jis"

    # CSV ファイルを読み込む
    try:
        with open(input_path, mode="r", encoding=input_encoding) as f:
            reader = csv.reader(f)
            rows = list(reader)
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding,
            e.object,
            e.start,
            e.end,
            f"ファイル '{input_path}' のデコードに失敗しました。"
            f"エラー: {e.reason}"
        )
    except FileNotFoundError:
        raise FileNotFoundError(
            f"入力ファイル '{input_path}' が見つかりません。"
        )

    # 出力ファイルを作成・書き込む
    try:
        with open(output_path, mode="w", encoding=encoding, newline="") as f:
            writer = csv.writer(f)
            writer.writerows(rows)
    except IOError as e:
        raise IOError(
            f"出力ファイル '{output_path}' の書き出しに失敗しました。"
            f"エラー: {e}"
        )

    print(f"✓ '{input_path}' → '{output_path}' の変換が完了しました。")


# --- 使用例 ---
if __name__ == "__main__":
    input_file = "data.csv"
    output_file = "data_utf8.csv"

    try:
        convert_csv(input_file, output_file)
    except FileNotFoundError as e:
        print(f"✗ エラー: {e}")
        sys.exit(1)
    except UnicodeDecodeError as e:
        print(f"✗ エラー: {e}")
        sys.exit(1)
    except IOError as e:
        print(f"✗ エラー: {e}")
        sys.exit(1)
