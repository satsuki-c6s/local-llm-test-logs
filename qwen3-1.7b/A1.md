### ✅ 函数：`convert_shift_jis_to_utf8`

```python
import codecs
import csv
import os

def convert_shift_jis_to_utf8(input_file, output_file):
    """
    Shift-JIS CSVファイルをUTF-8に変換し、出力ファイルを作成する。
    存在しないファイルやデコード失敗時のエラーハンドリングも含む。
    """
    try:
        # 入力ファイルが存在するか確認
        if not os.path.exists(input_file):
            raise FileNotFoundError(f"ファイルが存在しません: {input_file}")

        # 入力ファイルを開く（Shift-JISエンコーディング）
        with codecs.open(input_file, 'r', encoding='shift_jis') as infile:
            reader = csv.reader(infile)
            data = [row for row in reader]

        # 出力ファイルを作成（UTF-8エンコーディング）
        with codecs.open(output_file, 'w', encoding='utf-8') as outfile:
            writer = csv.writer(outfile)
            writer.writerows(data)

        print(f"成功: {output_file} に変換完了")

    except FileNotFoundError as e:
        print(f"エラー: {e}")
    except UnicodeDecodeError as e:
        print(f"デコードエラー: {e}")
    except Exception as e:
        print(f"予期せぬエラー: {e}")

    return
```
