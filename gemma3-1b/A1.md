import os

def shift_jis_to_utf8(input_file, output_file):
    """
    Shift-JISのCSVファイルをUTF-8で保存し直す関数。

    Args:
        input_file (str): 入力ファイル名（Shift-JIS形式）。
        output_file (str): 出力ファイル名（UTF-8形式）。

    Returns:
       bool: 成功した場合 True、失敗した場合 False。
    """
    try:
      with open(input_file, 'r', encoding='shift_jis') as infile:
          data = infile.read()  # ファイル全体を読み込む
        
      # 文字化けの確認
      if not data.strip(): #空ファイルの場合にエラーにならないようにする
         print("入力ファイルは空です。")
         return False

      try:
            with open(output_file, 'w', encoding='utf-8') as outfile:
              outfile.write(data)  # UTF-8で書き込む
        
          
      except Exception as e:
           print(f"ファイル書き込み中にエラーが発生しました：{e}") 

    except FileNotFoundError:
      print("指定されたファイルが見つかりません。")
      return False
    except UnicodeDecodeError:
       print("Shift-JISのデコードに失敗しました。")  # デコードエラーを処理
    except Exception as e: #その他の例外処理
        print(f"予期しないエラーが発生しました：{e}")
        return False

    return True


if __name__ == "__main__":
    input_file = "shift_jis.csv"
    output_file = "utf8_data.csv"

    # 例: Shift-JISファイルをUTF-8に変換する

    else:
