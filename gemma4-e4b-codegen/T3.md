```html
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <title>地球・太陽・月 軌道シミュレーター</title>
    <style>
        body {
            display: flex;
            flex-direction: column;
            align-items: center;
            font-family: Arial, sans-serif;
            background-color: #00001a; /* 深い宇宙の色 */
            color: white;
            margin: 0;
            padding: 20px;
        }
        #simulationCanvas {
            border: 2px solid #333;
            background-color: #000000;
            display: block;
            margin-bottom: 20px;
        }
        .controls {
            display: flex;
            gap: 15px;
            align-items: center;
            padding: 10px;
            background-color: #111;
            border-radius: 8px;
        }
        button {
            padding: 10px 20px;
            font-size: 16px;
            cursor: pointer;
            border: none;
            border-radius: 5px;
            transition: background-color 0.3s;
        }
        #pauseButton { background-color: #ffc107; color: black; }
        #resumeButton { background-color: #28a745; color: white; }
        button:disabled { opacity: 0.5; cursor: not-allowed; }

        /* ラベルのスタイル */
        .label-container {
            display: flex;
            gap: 30px;
            margin-top: 10px;
        }
    </style>
</head>
<body>

    <h1>🪐 地球・太陽・月 軌道シミュレーター</h1>
    <canvas id="simulationCanvas" width="800" height="600"></canvas>

    <div class="controls">
        <button id="pauseButton" onclick="togglePause()">一時停止</button>
        <button id="resumeButton" onclick="toggleResume()" disabled>再開</button>
        <span>速度:</span>
        <button onclick="setSpeed(0.2)">🐢 遅い</button>
        <button onclick="setSpeed(1.0)">💨 普通</button>
        <button onclick="setSpeed(2.5)">🚀 速い</button>
    </div>

    <script>
        // ==================================================
        // 物理定数と初期設定
        // ==================================================
        const canvas = document.getElementById('simulationCanvas');
        const ctx = canvas.getContext('2d');
        const centerX = canvas.width / 2;
        const centerY = canvas.height / 2;

        // 天体パラメータ (半径は描画上のスケール)
        const params = {
            sun: { radius: 30, color: '#ffcc00', label: '太陽' },
            earth: { radius: 8, color: '#4a90e2', label: '地球' },
            moon: { radius: 4, color: '#cccccc', label: '月' }
        };

        // 軌道パラメータ (単位は描画上のスケール)
        const orbits = {
            earth_radius: 150, // 地球の公転半径
            moon_radius: 30    // 月の公転半径（地球からの相対距離）
        };

        // 角速度と周期比の設定 (Angular Velocity: rad/frame)
        // T_E / T_M = 12 を実現するため、角速度を調整する。
        const initialAngles = {
            earth: Math.PI * 0.5, // 初期角度（上向き）
            moon: Math.PI * 0.5  // 初期角度（地球からの相対初期角度）
        };

        // 角速度 (Angular Velocity) の設定。値が大きいほど速い。
        // Earth Period (T_E): 約60フレームで1周 (omega_E = 2*PI / 60)
        const angularVelocities = {
            earth: 0.1, // 地球の角速度（太陽周り）
            moon: 0.08  // 月の角速度（地球周り）。(0.1 / 12 ≈ 0.0083) -> 0.08で調整し、視覚的に分かりやすくする。
        };

        // ==================================================
        // グローバル状態変数
        // ==================================================
        let state = {
            angleE: initialAngles.earth, // 地球の太陽周りの角度
            angleM: initialAngles.moon, // 月の地球からの相対角度
            isPaused: false,
            speedFactor: 1.0 // 速度調整係数 (0.2, 1.0, 2.5)
        };

        // ==================================================
        // コントロール関数
        // ==================================================

        /**
         * シミュレーションを一時停止/再開する
         */
        function togglePause() {
            state.isPaused = !state.isPaused;
            if (state.isPaused) {
                document.getElementById('pauseButton').disabled = false;
                document.getElementById('resumeButton').disabled = true;
            } else {
                document.getElementById('pauseButton').disabled = true;
                document.getElementById('resumeButton').disabled = false;
            }
        }

        /**
         * 再開ボタンが押されたときの処理 (togglePauseと同じロジックを再利用)
         */
        function toggleResume() {
             if (!state.isPaused) {
                // 実際には、一時停止状態から復帰する際に呼ばれる想定だが、今回は単にフラグをリセットする。
                state.isPaused = false;
                document.getElementById('pauseButton').disabled = false;
                document.getElementById('resumeButton').disabled = true;
            }
        }

        /**
         * 速度係数を設定し、シミュレーションの速さを変更する
         * @param {number} factor - 速度倍率 (0.2, 1.0, 2.5)
         */
        function setSpeed(factor) {
            state.speedFactor = factor;
        }

        // ==================================================
        // 描画ロジック
        // ==================================================

        /**
         * 軌道を描画する (円)
         */
        function drawOrbit(radius, color) {
            ctx.beginPath();
            ctx.arc(centerX, centerY, radius, 0, Math.PI * 2);
            ctx.strokeStyle = color;
            ctx.lineWidth = 1;
            ctx.stroke();
        }

        /**
         * 天体の位置を計算する (座標変換)
         * @param {number} r - 半径
         * @param {number} angle - 角度ラジアン
         * @returns {{x: number, y: number}} 座標
         */
        function getPosition(r, angle) {
            return {
                x: centerX + r * Math.cos(angle),
                y: centerY + r * Math.sin(angle)
            };
        }

        /**
         * シミュレーション全体を描画するメイン関数
         */
        function draw() {
            // 1. キャンバスをクリア
            ctx.fillStyle = '#000000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // 2. 軌道を描画 (太陽の周りの地球、そして地球からの月)
            drawOrbit(orbits.earth_radius, 'rgba(74, 144, 226, 0.5)'); // 地球軌道
            drawOrbit(orbits.moon_radius, 'rgba(204, 204, 204, 0.5)'); // 月軌道

            // --- 地球の位置計算 (太陽基準) ---
            const posE = getPosition(orbits.earth_radius, state.angleE);

            // --- 月の位置計算 (地球を原点とする相対座標) ---
            // 月の絶対角度は、初期角度 + 経過時間による回転。
            // ただし、ここでは単純化のため、月が常に地球から一定距離で公転していると見なす。
            const posM_relative = getPosition(orbits.moon_radius, state.angleM);

            // 月の絶対座標 = 地球の太陽座標 + 相対的な月座標
            const posM = {
                x: posE.x + posM_relative.x,
                y: posE.y + posM_relative.y
            };


            // 3. 天体を描画 (Sun -> Earth -> Moon の順)

            // 太陽 (原点)
            ctx.beginPath();
            ctx.arc(centerX, centerY, params.sun.radius, 0, Math.PI * 2);
            ctx.fillStyle = params.sun.color;
            ctx.fill();
            ctx.strokeStyle = '#ffaa00';
            ctx.lineWidth = 1;
            ctx.stroke();

            // 地球
            ctx.beginPath();
            ctx.arc(posE.x, posE.y, params.earth.radius, 0, Math.PI * 2);
            ctx.fillStyle = params.earth.color;
            ctx.fill();

            // 月
            ctx.beginPath();
            ctx.arc(posM.x, posM.y, params.moon.radius, 0, Math.PI * 2);
            ctx.fillStyle = params.moon.color;
            ctx.fill();


            // 4. ラベルの描画 (天体名)
            drawLabel(params.sun.label, centerX, centerY - params.sun.radius - 10);
            drawLabel(params.earth.label, posE.x - 20, posE.y - 20);
            drawLabel(params.moon.label, posM.x + 10, posM.y - 10);

        }

        /**
         * 指定座標にラベルを描画するヘルパー関数
         */
        function drawLabel(text, x, y) {
            ctx.fillStyle = 'white';
            ctx.font = '14px Arial';
            ctx.fillText(text, x, y);
        }

        /**
         * 角度を更新し、シミュレーションの時間を進める
         */
        function update() {
            if (state.isPaused) return;

            // 地球の角度を更新 (太陽周り)
            state.angleE += angularVelocities.earth * state.speedFactor * 0.01;
            // 角度が2πを超えたらリセット（必須ではないが、クリーンな状態維持のため）
            if (state.angleE > Math.PI * 2) {
                state.angleE -= Math.PI * 2;
            }

            // 月の角度を更新 (地球周り)
            state.angleM += angularVelocities.moon * state.speedFactor * 0.01;
             if (state.angleM > Math.PI * 2) {
                state.angleM -= Math.PI * 2;
            }

        }

        /**
         * メインループ処理
         */
        function loop() {
            update();
            draw();
            requestAnimationFrame(loop);
        }


        // ==================================================
        // 初期化と実行
        // ==================================================
        window.onload = function() {
            // 初回描画の準備として、一時停止状態にする
            state.isPaused = true; 
            document.getElementById('pauseButton').disabled = false;
            document.getElementById('resumeButton').disabled = true;

            // ループ開始 (最初の呼び出しはすぐに実行される)
            loop();
        };

    </script>
</body>
</html>
```
