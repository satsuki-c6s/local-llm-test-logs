import csv
import os


def convert_csv_encoding(input_path: str, output_path: str) -> None:
    """
    Shift-JISエンコーディングのCSVファイルを読み込み、UTF-8エンコーディングで保存し直す関数。
    
    Args:
        input_path: 入力CSVファイルのパス（Shift-JIS）
        output_path: 出力CSVファイルのパス（UTF-8）
    
    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: デコードに失敗した場合
    """
    # ファイルの存在確認
    if not os.path.exists(input_path):
        raise FileNotFoundError(f"ファイルが見つかりません: {input_path}")
    
    try:
        # Shift-JISで読み込み
        with open(input_path, 'r', encoding='shift_jis') as f_in:
            reader = csv.reader(f_in)
            rows = list(reader)
        
        # UTF-8で書き出し
        with open(output_path, 'w', encoding='utf-8', newline='') as f_out:
            writer = csv.writer(f_out)
            writer.writerows(rows)
    
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding,
            e.object,
            e.start,
            e.end,
            f"Shift-JISからのデコードに失敗しました: {e}"
        )
