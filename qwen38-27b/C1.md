## SQLAlchemy 基本の使い方

### 1. インストール

```bash
pip install sqlalchemy
```

### 2. モデル定義と接続

```python
from sqlalchemy import create_engine, Column, Integer, String, Float
from sqlalchemy.orm import declarative_base, Session

# Baseクラスを生成
Base = declarative_base()

# モデル定義
class User(Base):
    __tablename__ = 'users'

    id = Column(Integer, primary_key=True, autoincrement=True)
    name = Column(String(100), nullable=False)
    email = Column(String(255), unique=True, nullable=False)
    age = Column(Integer, nullable=True)

    def __repr__(self):
        return f"<User(id={self.id}, name='{self.name}', email='{self.email}')>"


class Post(Base):
    __tablename__ = 'posts'

    id = Column(Integer, primary_key=True, autoincrement=True)
    title = Column(String(200), nullable=False)
    body = Column(String, nullable=True)
    user_id = Column(Integer, nullable=False)  # 外部キー

    def __repr__(self):
        return f"<Post(id={self.id}, title='{self.title}')>"


# データベース接続（SQLiteの例）
engine = create_engine('sqlite:///example.db', echo=True)

# テーブル作成
Base.metadata.create_all(engine)
```

### 3. CRUD操作

```python
from sqlalchemy.orm import Session

# セッション開始
with Session(engine) as session:
    # ===== Create =====
    user = User(name='田中太郎', email='taro@example.com', age=30)
    session.add(user)
    session.commit()
    print(f"ユーザー作成: {user.id}")

    # 複数件追加
    users = [
        User(name='鈴木花子', email='hanako@example.com', age=25),
        User(name='高橋次郎', email='jiro@example.com', age=35),
    ]
    session.add_all(users)
    session.commit()

    # ===== Read =====
    # 1件取得
    user = session.get(User, 1)
    print(f"取得: {user.name}, {user.email}")

    # 条件付き検索
    young_users = session.query(User).filter(User.age < 30).all()
    for u in young_users:
        print(f"  {u.name} ({u.age}歳)")

    # 件数取得
    count = session.query(User).count()
    print(f"ユーザー総数: {count}")

    # ===== Update =====
    user = session.get(User, 1)
    user.age = 31
    user.name = '田中太郎（更新）'
    session.commit()
    print(f"更新: {user.name}, {user.age}歳")

    # ===== Delete =====
    user = session.get(User, 2)
    session.delete(user)
    session.commit()
    print("削除完了")
```

### 4. リレーションシップ（1対多）

```python
from sqlalchemy import ForeignKey
from sqlalchemy.orm import relationship

class User(Base):
    __tablename__ = 'users'

    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)
    email = Column(String(255), unique=True, nullable=False)

    # リレーションシップ定義
    posts = relationship("Post", back_populates="author", cascade="all, delete-orphan")

    def __repr__(self):
        return f"<User(id={self.id}, name='{self.name}')>"


class Post(Base):
    __tablename__ = 'posts'

    id = Column(Integer, primary_key=True)
    title = Column(String(200), nullable=False)
    body = Column(String, nullable=True)
    user_id = Column(Integer, ForeignKey('users.id'), nullable=False)

    # リレーションシップ定義
    author = relationship("User", back_populates="posts")

    def __repr__(self):
        return f"<Post(id={self.id}, title='{self.title}')>"


# 使用例
with Session(engine) as session:
    # ユーザーと投稿を作成
    user = User(name='田中太郎', email='taro@example.com')
    post1 = Post(title='はじめての投稿', body='こんにちは！', author=user)
    post2 = Post(title='2つ目の投稿', body='また投稿しました。', author=user)

    session.add(user)
    session.commit()

    # リレーションシップ経由でアクセス
    print(f"{user.name} の投稿:")
    for p in user.posts:
        print(f"  - {p.title}")

    # 逆方向アクセス
    print(f"投稿 '{post1.title}' の著者: {post1.author.name}")
```

### 5. 高度なクエリ

```python
from sqlalchemy import func, or_

with Session(engine) as session:
    # 集計関数
    avg_age = session.query(func.avg(User.age)).scalar()
    print(f"平均年齢: {avg_age}")

    # GROUP BY
    results = session.query(User.name, func.count(Post.id).label('post_count')) \
                     .outerjoin(Post) \
                     .group_by(User.id) \
                     .all()
    for name, count in results:
        print(f"  {name}: {count}件の投稿")

    # OR条件
    users = session.query(User).filter(
        or_(User.name.like('%田中%'), User.age > 30)
    ).all()

    # ソート
    users = session.query(User).order_by(User.age.desc()).all()

    # LIMIT / OFFSET
    users = session.query(User).limit(5).offset(10).all()
```

### 6. トランザクション管理

```python
with Session(engine) as session:
    try:
        user = User(name='テスト', email='test@example.com')
        session.add(user)

        post = Post(title='テスト投稿', body='本文', author=user)
        session.add(post)

        session.commit()  # 両方ともコミット
        print("成功: ユーザーと投稿が作成されました")
    except Exception as e:
        session.rollback()  # エラー時はロールバック
        print(f"失敗: {e}")
```
