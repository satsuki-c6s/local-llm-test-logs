import csv
import os
import sys


def convert_shift_jis_to_utf8(input_path: str, output_path: str) -> None:
    """
    Shift-JISエンコードのCSVファイルを読み込み、UTF-8エンコードで保存し直す。

    Args:
        input_path: 入力ファイルのパス（Shift-JISエンコードのCSV）
        output_path: 出力ファイルのパス（UTF-8エンコードのCSV）

    Raises:
        FileNotFoundError: 入力ファイルが存在しない場合
        UnicodeDecodeError: Shift-JISとしてデコードできない場合
        IOError: ファイル読み書き中にエラーが発生した場合
    """
    # ファイル存在チェック
    if not os.path.exists(input_path):
        raise FileNotFoundError(f"入力ファイルが存在しません: {input_path}")

    # 出力ディレクトリが存在しない場合は作成
    output_dir = os.path.dirname(output_path)
    if output_dir and not os.path.exists(output_dir):
        os.makedirs(output_dir, exist_ok=True)

    # Shift-JISで読み込み、UTF-8で書き出し
    try:
        with open(input_path, 'r', encoding='shift_jis', newline='') as infile:
            reader = csv.reader(infile)
            rows = list(reader)
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding, e.object, e.start, e.end,
            f"Shift-JISとしてデコードできません。ファイルがShift-JISエンコードでない可能性があります: {input_path}"
        ) from e
    except IOError as e:
        raise IOError(f"ファイルの読み込みに失敗しました: {input_path}") from e

    try:
        with open(output_path, 'w', encoding='utf-8', newline='') as outfile:
            writer = csv.writer(outfile)
            writer.writerows(rows)
    except IOError as e:
        raise IOError(f"ファイルの書き込みに失敗しました: {output_path}") from e


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("使用方法: python convert_csv.py <入力ファイル> <出力ファイル>")
        sys.exit(1)

    input_file = sys.argv[1]
    output_file = sys.argv[2]

    try:
        convert_shift_jis_to_utf8(input_file, output_file)
        print(f"変換完了: {input_file} -> {output_file}")
    except FileNotFoundError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
    except UnicodeDecodeError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
    except IOError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
