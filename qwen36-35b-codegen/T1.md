<script>
(function() {
  const STORAGE_KEY = "todo_tasks_v1";
  let tasks = loadTasks();
  let currentFilter = "all";

  const taskInput   = document.getElementById("taskInput");
  const addBtn      = document.getElementById("addBtn");
  const taskList    = document.getElementById("taskList");
  const footer      = document.getElementById("footer");
  const filterBtns  = document.querySelectorAll(".filter-btn");

  /* ── localStorage ── */
  function loadTasks() {
    try { return JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; }
    catch { return []; }
  }
  function saveTasks() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
  }

  /* ── 追加 ── */
  function addTask(text) {
    text = text.trim();
    if (!text) return;
    tasks.push({ id: Date.now(), text, completed: false });
    saveTasks();
    render();
    taskInput.value = "";
    taskInput.focus();
  }

  addBtn.addEventListener("click", () => addTask(taskInput.value));
  taskInput.addEventListener("keydown", e => {
    if (e.key === "Enter") addTask(taskInput.value);
  });

  /* ── フィルタ ── */
  filterBtns.forEach(btn => {
    btn.addEventListener("click", () => {
      filterBtns.forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      currentFilter = btn.dataset.filter;
      render();
    });
  });

  function getFiltered() {
    if (currentFilter === "active")   return tasks.filter(t => !t.completed);
    if (currentFilter === "completed") return tasks.filter(t => t.completed);
    return tasks;
  }

  /* ── 完了切り替え ── */
  function toggleTask(id) {
    const task = tasks.find(t => t.id === id);
    if (task) task.completed = !task.completed;
    saveTasks();
    render();
  }

  /* ── 削除 ── */
  function deleteTask(id) {
    const el = document.querySelector(`[data-id="${id}"]`);
    if (el) {
      el.classList.add("removing");
      setTimeout(() => {
        tasks = tasks.filter(t => t.id !== id);
        saveTasks();
        render();
      }, 300);
    } else {
      tasks = tasks.filter(t => t.id !== id);
      saveTasks();
      render();
    }
  }

  /* ── 編集（ダブルクリック） ── */
  function startEdit(id) {
    const task = tasks.find(t => t.id === id);
    if (!task) return;
    const li = document.querySelector(`[data-id="${id}"]`);
    const textSpan = li.querySelector(".task-text");

    const input = document.createElement("input");
    input.type = "text";
    input.className = "edit-input";
    input.value = task.text;
    textSpan.replaceWith(input);
    input.focus();
    input.select();

    function finishEdit() {
      const newText = input.value.trim();
      if (newText) {
        task.text = newText;
        saveTasks();
      }
      render();
    }

    input.addEventListener("blur", finishEdit);
    input.addEventListener("keydown", e => {
      if (e.key === "Enter") input.blur();
      if (e.key === "Escape") { input.value = task.text; input.blur(); }
    });
  }

  /* ── レンダリング ── */
  function render() {
    const filtered = getFiltered();
    const remaining = tasks.filter(t => !t.completed).length;

    if (tasks.length === 0) {
      taskList.innerHTML = `
        <div class="empty-state">
          <div class="icon">🎉</div>
          <p>タスクがありません<br>追加してみましょう！</p>
        </div>`;
      footer.textContent = "";
      return;
    }

    if (filtered.length === 0) {
      taskList.innerHTML = `
        <div class="empty-state">
          <div class="icon">🔍</div>
          <p>該当するタスクはありません</p>
        </div>`;
      footer.innerHTML = `<span>${remaining}件が未完了です</span>`;
      return;
    }

    taskList.innerHTML = filtered.map(task => `
      <li class="task-item ${task.completed ? 'completed' : ''}" data-id="${task.id}">
        <div class="task-checkbox" title="完了切り替え"></div>
        <span class="task-text">${escapeHtml(task.text)}</span>
        <button class="delete-btn" title="削除">✕</button>
      </li>
    `).join("");

    /* イベント設定 */
    taskList.querySelectorAll(".task-checkbox").forEach(cb => {
      cb.addEventListener("click", () => toggleTask(Number(cb.closest(".task-item").dataset.id)));
    });
    taskList.querySelectorAll(".delete-btn").forEach(btn => {
      btn.addEventListener("click", () => deleteTask(Number(btn.closest(".task-item").dataset.id)));
    });
    taskList.querySelectorAll(".task-text").forEach(span => {
      span.addEventListener("dblclick", () => startEdit(Number(span.closest(".task-item").dataset.id)));
    });

    footer.innerHTML = `<span>${remaining}件が未完了です</span>`;
  }

  function escapeHtml(str) {
    const div = document.createElement("div");
    div.textContent = str;
    return div.innerHTML;
  }

  /* ── 初期表示 ── */
  render();
})();
</script>
</body>

## 機能まとめ

| # | 要件 | 実装方法 |
|---|------|----------|
| 1 | タスク追加 | テキスト入力 + 「追加」ボタン / Enterキー |
| 2 | 完了切り替え | チェックボックス（丸）をクリック |
| 3 | タスク削除 | 右端の ✕ ボタン（アニメーション付き） |
| 4 | インライン編集 | タスクテキストを**ダブルクリック**で編集モードに |
| 5 | フィルタ | 「すべて / 未完了 / 完了」の3ボタン切り替え |
| 6 | 未完了件数表示 | フッターに「〇件が未完了です」と表示 |
| 7 | localStorage永続化 | `localStorage` に JSON で保存・復元 |

**追加ポイント:**
- エスケープ処理でXSS対策済み
- 削除時にスライドアウトアニメーション
- 編集時は Enter で確定、Escape でキャンセル
- グラデーション背景 + カードUIのモダンデザイン
