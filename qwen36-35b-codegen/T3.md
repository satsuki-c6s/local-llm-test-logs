<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');

// --- Canvas sizing ---
function resize() {
  const s = Math.min(window.innerWidth - 40, window.innerHeight - 160, 700);
  canvas.width = s;
  canvas.height = s;
}
resize();
window.addEventListener('resize', () => { resize(); });

// --- Simulation state ---
const sunRadius = 30;
const earthOrbitRadius = 200;
const moonOrbitRadius = 55;
const earthPeriod = 6000;       // ms for one Earth orbit at speed=1
const moonPeriod = earthPeriod / 12; // Moon orbits ~12 times per Earth orbit

let time = 0;
let paused = false;
let speedMultiplier = 1;
let lastTimestamp = null;

// --- Stars background ---
const stars = [];
for (let i = 0; i < 300; i++) {
  stars.push({
    x: Math.random(),
    y: Math.random(),
    r: Math.random() * 1.5 + 0.3,
    a: Math.random() * 0.6 + 0.2
  });
}

function drawStars() {
  for (const s of stars) {
    ctx.beginPath();
    ctx.arc(s.x * canvas.width, s.y * canvas.height, s.r, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(255,255,255,${s.a})`;
    ctx.fill();
  }
}

function drawSun(cx, cy) {
  // Outer glow
  const grd3 = ctx.createRadialGradient(cx, cy, sunRadius * 0.5, cx, cy, sunRadius * 4);
  grd3.addColorStop(0, 'rgba(255,200,50,0.15)');
  grd3.addColorStop(1, 'rgba(255,200,50,0)');
  ctx.beginPath();
  ctx.arc(cx, cy, sunRadius * 4, 0, Math.PI * 2);
  ctx.fillStyle = grd3;
  ctx.fill();

  // Mid glow
  const grd2 = ctx.createRadialGradient(cx, cy, sunRadius * 0.8, cx, cy, sunRadius * 2.2);
  grd2.addColorStop(0, 'rgba(255,160,20,0.3)');
  grd2.addColorStop(1, 'rgba(255,160,20,0)');
  ctx.beginPath();
  ctx.arc(cx, cy, sunRadius * 2.2, 0, Math.PI * 2);
  ctx.fillStyle = grd2;
  ctx.fill();

  // Sun body
  const grd = ctx.createRadialGradient(cx - sunRadius * 0.25, cy - sunRadius * 0.25, sunRadius * 0.1, cx, cy, sunRadius);
  grd.addColorStop(0, '#fff8e0');
  grd.addColorStop(0.3, '#ffe44d');
  grd.addColorStop(0.7, '#ff9900');
  grd.addColorStop(1, '#cc6600');
  ctx.beginPath();
  ctx.arc(cx, cy, sunRadius, 0, Math.PI * 2);
  ctx.fillStyle = grd;
  ctx.fill();

  // Corona rays
  ctx.save();
  ctx.globalAlpha = 0.12 + 0.04 * Math.sin(time / 800);
  for (let i = 0; i < 12; i++) {
    const angle = (i / 12) * Math.PI * 2 + time / 5000;
    const len = sunRadius * (1.6 + 0.3 * Math.sin(time / 600 + i));
    ctx.beginPath();
    ctx.moveTo(cx, cy);
    ctx.lineTo(
      cx + Math.cos(angle) * len,
      cy + Math.sin(angle) * len
    );
    ctx.strokeStyle = '#ffcc44';
    ctx.lineWidth = 2;
    ctx.stroke();
  }
  ctx.restore();

  // Label
  ctx.fillStyle = '#ffe066';
  ctx.font = `bold ${Math.max(12, canvas.width * 0.02)}px 'Segoe UI', sans-serif`;
  ctx.textAlign = 'center';
  ctx.fillText('太陽 (Sun)', cx, cy + sunRadius + 18);
}

function drawEarth(ex, ey) {
  const er = 14;

  // Atmosphere glow
  const grd = ctx.createRadialGradient(ex, ey, er * 0.8, ex, ey, er * 2.5);
  grd.addColorStop(0, 'rgba(80,160,255,0.2)');
  grd.addColorStop(1, 'rgba(80,160,255,0)');
  ctx.beginPath();
  ctx.arc(ex, ey, er * 2.5, 0, Math.PI * 2);
  ctx.fillStyle = grd;
  ctx.fill();

  // Earth body
  const grd2 = ctx.createRadialGradient(ex - er * 0.3, ey - er * 0.3, er * 0.1, ex, ey, er);
  grd2.addColorStop(0, '#6db3f2');
  grd2.addColorStop(0.4, '#2e8bc0');
  grd2.addColorStop(0.7, '#1a6e9e');
  grd2.addColorStop(1, '#0d4a6b');
  ctx.beginPath();
  ctx.arc(ex, ey, er, 0, Math.PI * 2);
  ctx.fillStyle = grd2;
  ctx.fill();

  // Land masses (simplified)
  ctx.save();
  ctx.beginPath();
  ctx.arc(ex, ey, er, 0, Math.PI * 2);
  ctx.clip();
  const landAngle = time / earthPeriod * Math.PI * 2;
  for (let i = 0; i < 5; i++) {
    const la = landAngle + i * 1.3;
    const lx = ex + Math.cos(la) * er * 0.4;
    const ly = ey + Math.sin(la) * er * 0.35;
    ctx.beginPath();
    ctx.arc(lx, ly, er * 0.35, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(60,160,80,0.45)';
    ctx.fill();
  }
  // Ice cap
  ctx.beginPath();
  ctx.arc(ex, ey - er * 0.7, er * 0.3, 0, Math.PI * 2);
  ctx.fillStyle = 'rgba(255,255,255,0.3)';
  ctx.fill();
  ctx.restore();

  // Label
  ctx.fillStyle = '#88ccff';
  ctx.font = `bold ${Math.max(11, canvas.width * 0.017)}px 'Segoe UI', sans-serif`;
  ctx.textAlign = 'center';
  ctx.fillText('地球 (Earth)', ex, ey + er + 16);
}

function drawMoon(mx, my) {
  const mr = 6;

  // Moon body
  const grd = ctx.createRadialGradient(mx - mr * 0.3, my - mr * 0.3, mr * 0.1, mx, my, mr);
  grd.addColorStop(0, '#e8e8e8');
  grd.addColorStop(0.6, '#c0c0c0');
  grd.addColorStop(1, '#888');
  ctx.beginPath();
  ctx.arc(mx, my, mr, 0, Math.PI * 2);
  ctx.fillStyle = grd;
  ctx.fill();

  // Craters
  ctx.save();
  ctx.beginPath();
  ctx.arc(mx, my, mr, 0, Math.PI * 2);
  ctx.clip();
  const craters = [[-1.5, -1], [2, 1], [-0.5, 2.5]];
  for (const [cx2, cy2] of craters) {
    ctx.beginPath();
    ctx.arc(mx + cx2, my + cy2, mr * 0.25, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(100,100,100,0.35)';
    ctx.fill();
  }
  ctx.restore();

  // Label
  ctx.fillStyle = '#bbb';
  ctx.font = `${Math.max(10, canvas.width * 0.014)}px 'Segoe UI', sans-serif`;
  ctx.textAlign = 'center';
  ctx.fillText('月 (Moon)', mx, my + mr + 13);
}

function drawOrbitPath(cx, cy, radius, color, dash) {
  ctx.beginPath();
  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
  ctx.strokeStyle = color;
  ctx.lineWidth = 1;
  if (dash) ctx.setLineDash(dash);
  else ctx.setLineDash([]);
  ctx.stroke();
  ctx.setLineDash([]);
}

function drawMoonTrail(ex, ey, moonAngle) {
  // Draw a small arc showing the moon's current position relative to Earth
  const trailLen = Math.PI * 0.6;
  const startAngle = moonAngle - trailLen / 2;
  ctx.beginPath();
  ctx.arc(ex, ey, moonOrbitRadius, startAngle, moonAngle);
  ctx.strokeStyle = 'rgba(200,200,200,0.15)';
  ctx.lineWidth = 1;
  ctx.setLineDash([3, 4]);
  ctx.stroke();
  ctx.setLineDash([]);
}

function drawEarthTrail(cx, cy, earthAngle) {
  const trailLen = Math.PI * 0.15;
  const startAngle = earthAngle - trailLen / 2;
  ctx.beginPath();
  ctx.arc(cx, cy, earthOrbitRadius, startAngle, earthAngle);
  ctx.strokeStyle = 'rgba(80,160,255,0.15)';
  ctx.lineWidth = 1;
  ctx.setLineDash([3, 4]);
  ctx.stroke();
  ctx.setLineDash([]);
}

function drawInfoPanel(cx, cy) {
  const daysElapsed = (time / earthPeriod) * 365;
  const moonOrbits = Math.floor((time / moonPeriod));

  ctx.fillStyle = 'rgba(0,0,0,0.5)';
  const pw = 180, ph = 52;
  const px = canvas.width - pw - 12, py = 12;
  ctx.beginPath();
  ctx.roundRect(px, py, pw, ph, 6);
  ctx.fill();

  ctx.fillStyle = '#aaa';
  ctx.font = `${Math.max(10, canvas.width * 0.013)}px 'Segoe UI', sans-serif`;
  ctx.textAlign = 'left';
  ctx.fillText(`経過日数: ${daysElapsed.toFixed(1)}日`, px + 12, py + 20);
  ctx.fillText(`月の公転回数: ${moonOrbits}回`, px + 12, py + 40);
}

function draw() {
  const W = canvas.width;
  const H = canvas.height;
  const cx = W / 2;
  const cy = H / 2;

  ctx.clearRect(0, 0, W, H);

  // Background gradient
  const bgGrd = ctx.createRadialGradient(cx, cy, 50, cx, cy, W * 0.7);
  bgGrd.addColorStop(0, '#0f0f2e');
  bgGrd.addColorStop(1, '#050510');
  ctx.fillStyle = bgGrd;
  ctx.fillRect(0, 0, W, H);

  drawStars();

  // Orbit paths
  drawOrbitPath(cx, cy, earthOrbitRadius, 'rgba(80,160,255,0.2)', [4, 6]);
  drawOrbitPath(cx, cy, moonOrbitRadius + earthOrbitRadius, 'rgba(200,200,200,0.06)', [2, 8]);

  // Compute positions
  const earthAngle = (time / earthPeriod) * Math.PI * 2;
  const moonAngle = (time / moonPeriod) * Math.PI * 2;

  const ex = cx + Math.cos(earthAngle) * earthOrbitRadius;
  const ey = cy + Math.sin(earthAngle) * earthOrbitRadius;
  const mx = ex + Math.cos(moonAngle) * moonOrbitRadius;
  const my = ey + Math.sin(moonAngle) * moonOrbitRadius;

  // Trails
  drawEarthTrail(cx, cy, earthAngle);
  drawMoonTrail(ex, ey, moonAngle);

  // Draw orbit line from Earth to Moon
  ctx.beginPath();
  ctx.moveTo(ex, ey);
  ctx.lineTo(mx, my);
  ctx.strokeStyle = 'rgba(200,200,200,0.1)';
  ctx.lineWidth = 1;
  ctx.stroke();

  // Draw bodies (back to front: sun, earth orbit line, moon, earth, sun glow on top)
  drawOrbitPath(cx, cy, earthOrbitRadius, 'rgba(80,160,255,0.2)', [4, 6]);

  drawSun(cx, cy);
  drawEarth(ex, ey);
  drawMoon(mx, my);

  // Info panel
  drawInfoPanel(cx, cy);

  // Pause overlay
  if (paused) {
    ctx.fillStyle = 'rgba(0,0,0,0.35)';
    ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${Math.max(20, canvas.width * 0.03)}px 'Segoe UI', sans-serif`;
    ctx.textAlign = 'center';
    ctx.fillText('⏸ 一時停止中', W / 2, H / 2);
  }
}

function animate(timestamp) {
  if (lastTimestamp === null) lastTimestamp = timestamp;
  const dt = timestamp - lastTimestamp;
  lastTimestamp = timestamp;

  if (!paused) {
    time += dt * speedMultiplier;
  }

  draw();
  requestAnimationFrame(animate);
}

// --- Controls ---
function togglePause() {
  paused = !paused;
  const btn = document.getElementById('pauseBtn');
  btn.textContent = paused ? '▶ 再開' : '⏸ 一時停止';
}

function setSpeed(s) {
  speedMultiplier = s;
  document.querySelectorAll('.controls button').forEach(b => b.classList.remove('active'));
  if (s === 0.3) document.getElementById('slowBtn').classList.add('active');
  else if (s === 1) document.getElementById('normalBtn').classList.add('active');
  else document.getElementById('fastBtn').classList.add('active');
}

// --- Start ---
requestAnimationFrame(animate);
</script>

</body>
</html>
