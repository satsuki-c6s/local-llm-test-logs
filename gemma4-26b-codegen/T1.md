<script>
    /**
     * JavaScript: ロジックの定義
     */

    // --- 状態管理 ---
    let todos = JSON.parse(localStorage.getItem('todos')) || [];
    let currentFilter = 'all'; // 'all', 'active', 'completed'

    const todoInput = document.getElementById('todo-input');
    const addBtn = document.getElementById('add-btn');
    const todoList = document.getElementById('todo-list');
    const activeCountText = document.getElementById('active-count');
    const filterBtns = document.querySelectorAll('.filter-btn');

    // --- メイン関数 ---

    // データの保存
    function saveToLocalStorage() {
        localStorage.setItem('todos', JSON.stringify(todos));
    }

    // タスクの追加
    function addTodo() {
        const text = todoInput.value.trim();
        if (text === '') return;

        const newTodo = {
            id: Date.now(),
            text: text,
            completed: false
        };

        todos.push(newTodo);
        todoInput.value = '';
        saveToLocalStorage();
        render();
    }

    // タスクの削除
    function deleteTodo(id) {
        todos = todos.filter(todo => todo.id !== id);
        saveToLocalStorage();
        render();
    }

    // タスクの完了切り替え
    function toggleTodo(id) {
        todos = todos.map(todo => {
            if (todo.id === id) {
                return { ...todo, completed: !todo.completed };
            }
            return todo;
        });
        saveToLocalStorage();
        render();
    }

    // タスクの編集（ダブルクリック時）
    function editTodo(id, currentText) {
        const li = document.querySelector(`li[data-id="${id}"]`);
        const span = li.querySelector('.todo-text');
        
        // 入力用インプットに差し替え
        const input = document.createElement('input');
        input.type = 'text';
        input.className = 'edit-input';
        input.value = currentText;
        
        li.replaceChild(input, span);
        input.focus();

        // 確定処理 (Enter または Blur)
        const finishEdit = () => {
            const newText = input.value.trim();
            if (new<0xA0>newText !== '') {
                todos = todos.map(todo => todo.id === id ? { ...todo, text: newText } : todo);
                saveToLocalStorage();
            }
            render();
        };

        input.addEventListener('blur', finishEdit);
        input.addEventListener('keydown', (e) => {
            if (e.key === 'Enter') finishEdit();
            if (e.key === 'Escape') render(); // キャンセル
        });
    }

    // 画面の描画（再レンダリング）
    function render() {
        // リストをクリア
        todoList.innerHTML = '';

        // フィルタリング
        const filteredTodos = todos.filter(todo => {
            if (currentFilter === 'active') return !todo.completed;
            if (currentFilter === 'completed') return todo.completed;
            return true; // 'all' の場合
        });

        // 各タスクのDOM生成
        filteredTodos.forEach(todo => {
            const li = document.createElement('li');
            li.setAttribute('data-id', todo.id);

            // チェックボックス
            const checkbox = document.createElement('input');
            checkbox.type = 'checkbox';
            checkbox.checked = todo.completed;
            checkbox.addEventListener('change', () => toggleTodo(todo.id));

            // テキスト部分
            const span = document.createElement('span');
            span.className = `todo-text ${todo.completed ? 'completed' : ''}`;
            span.textContent = todo.text;
            span.addEventListener('dblclick', () => editTodo(todo.id, todo.text));

            // 削除ボタン
            const delBtn = document.createElement('button');
            delBtn.className = 'delete-btn';
            delBtn.textContent = '削除';
            delBtn.addEventListener('click', () => deleteTodo(todo.id));

            li.appendChild(checkbox);
            li.appendChild(span);
            li.appendChild(delBtn);
            todoList.appendChild(li);
        });

        // 未完了件数の更新
        const activeCount = todos.filter(t => !t.completed).length;
        active/activeCountText.textContent = `未完了: ${activeCount}`;

        // フィルタボタンの見た目更新
        filterBtns.forEach(btn => {
            if (btn.getAttribute('data-filter') === currentFilter) {
                btn.classList.add('active');
            } else {
                btn.classList.remove('active');
            }
        });
    }

    // --- イベントリスナーの設定 ---

    todoInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') addTodo();
    });

    // フィルタ切り替え
    filterBtns.forEach(btn => {
        btn.addEventListener('click', () => {
            currentFilter = btn.getAttribute('data-filter');
            render();
        });
    });

    // 初期表示
    render();

</script>

### 実装のポイント:

1.  **単一ファイル構成**: HTML, CSS, JavaScriptをすべて一つのファイルにまとめています。外部への依存（CDN等）は一切ありません。
2.  **状態管理 (State Management)**: `todos` という配列が唯一の真実（Single Source of Truth）であり、データの変更（追加・削除・編集・完了切り替え）が発生するたびに `render()` 関数を呼び出して、画面全体を現在のデータに基づいて再描画する仕組みにしています。これにより、フィルタリングや件数表示の同期が非常に簡単かつ正確になります。
3.  **編集機能**: タスクのテキスト部分をダブルクリックすると、その要素が `<input>` 要素に動的に置き換わります。Enterキーで保存、Escキーでキャンセル（再描画）ができるように実装しています。
4.  **永続化**: `localStorage` を使用して、ブラウザを閉じたりリロードしたりしてもデータが保持されるようにしています。
5.  **フィルタリング**: 「すべて」「未完了」「完了」の各状態に応じて、表示する配列を `filter()` メソッドで抽出して描画しています。
