import csv
import os
import sys


def convert_csv_encoding(input_path: str, output_path: str) -> None:
    """
    Shift-JISでエンコードされたCSVファイルを読み込み、UTF-8で保存し直す。
    
    Args:
        input_path: 入力CSVファイルのパス（Shift-JISエンコード）
        output_path: 出力CSVファイルのパス（UTF-8エンコード）
    
    Raises:
        FileNotFoundError: 入力ファイルが存在しない場合
        UnicodeDecodeError: 文字化け（デコード失敗）が発生した場合
        Exception: その他のエラーが発生した場合
    """
    # 入力ファイルの存在確認
    if not os.path.exists(input_path):
        raise FileNotFoundError(f"入力ファイルが見つかりません: {input_path}")
    
    if not os.path.isfile(input_path):
        raise FileNotFoundError(f"指定されたパスはファイルではありません: {input_path}")
    
    try:
        # Shift-JISで読み込み
        with open(input_path, 'r', encoding='shift-jis', newline='') as infile:
            reader = csv.reader(infile)
            rows = list(reader)
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding,
            e.object,
            e.start,
            e.end,
            f"Shift-JISデコードに失敗しました（文字化けの可能性）: {input_path}"
        ) from e
    except Exception as e:
        raise Exception(f"ファイル読み込み中にエラーが発生しました: {e}") from e
    
    try:
        # UTF-8で書き込み
        with open(output_path, 'w', encoding='utf-8', newline='') as outfile:
            writer = csv.writer(outfile)
            writer.writerows(rows)
    except Exception as e:
        raise Exception(f"ファイル書き込み中にエラーが発生しました: {e}") from e


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("使い方: python script.py <入力ファイル> <出力ファイル>")
        sys.exit(1)
    
    input_file = sys.argv[1]
    output_file = sys.argv[2]
    
    try:
        convert_csv_encoding(input_file, output_file)
        print(f"変換完了: {input_file} -> {output_file}")
    except FileNotFoundError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
    except UnicodeDecodeError as e:
        print(f"文字化けエラー: {e}", file=sys.stderr)
        sys.exit(1)
    except Exception as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
