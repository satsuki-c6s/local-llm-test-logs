```python
import csv

def convert_shift_jis_csv_to_utf8(input_file, output_file):
    """
    Shift-JISでエンコードされたCSVファイルを読み込み、UTF-8で保存し直す関数です。

    Args:
        input_file (str): 入力ファイルパス（Shift-JIS CSV）
        output_file (str): 出力ファイルパス（UTF-8 CSV）

    Returns:
        bool: 成功した場合はTrue、失敗した場合はFalseを返します。
    """
    try:
        with open(input_file, 'r', encoding='shift_jis') as infile, \
             open(output_file, 'w', encoding='utf-8', newline='') as outfile:

            reader = csv.reader(infile)
            writer = csv.writer(outfile)

            for row in reader:
                writer.writerow(row)  # 各行をUTF-8で書き込む

        return True

    except FileNotFoundError:
        print(f"エラー：ファイル '{input_file}' が見つかりません。")
        return False
    except UnicodeDecodeError as e:
        print(f"エラー：Shift-JISデコードに失敗しました。ファイル '{input_file}' のエンコーディングがShift-JISではない可能性があります。\n詳細: {e}")
        return False
    except Exception as e:  # その他のエラーをキャッチ
        print(f"予期せぬエラーが発生しました：{e}")
        return False


if __name__ == '__main__':
    input_csv_file = 'input.csv'  # 変換するShift-JIS CSVファイルのパス
    output_csv_file = 'output.csv' # UTF-8で保存するCSVファイルのパス

    success = convert_shift_jis_csv_to_utf8(input_csv_file, output_csv_file)

    if success:
        print(f"ファイル '{input_csv_file}' をUTF-8の '{output_csv_file}' に変換しました。")
    else:
        print("変換に失敗しました。エラーメッセージを確認してください。")
```

     - **`UnicodeDecodeError`:**  Shift-JISでデコードできない文字が含まれている場合にエラーメッセージを表示し、`False`を返します。  これは、ファイルが実際にShift-JISでエンコードされていない場合によく発生します。

**注意点:**

* **エンコーディングの確認:** 入力ファイルが本当にShift-JISでエンコードされていることを確認してください。もし違うエンコーディングであれば、`encoding='shift_jis'` の部分を適切なエンコーディングに変更する必要があります (例: `encoding='iso-8859-1'`, `encoding='cp932'`) 。
* **文字化け:**  Shift-JISでデコードできない文字が含まれている場合、`UnicodeDecodeError` が発生します。 この場合は、入力ファイルのエンコーディングが間違っているか、またはファイルに不正な文字が含まれている可能性があります。
* **改行コード:** `newline=''` は、特にWindows環境でCSVファイルを扱う場合に重要です。 これを指定しないと、余分な空行が挿入されることがあります。
* **エラー処理の強化:** より詳細なエラーハンドリングが必要な場合は、`except` ブロック内でより具体的な例外をキャッチしたり、エラーログを出力したりすることができます。
* **大きなファイル:** 非常に大きなファイルを扱う場合は、メモリ使用量を考慮して、少しずつ読み込んで書き込む方法 (チャンク処理) を検討してください。
