import csv
import sys
import codecs

def convert_shift_jis_to_utf8(input_path, output_path):
    """
    Shift-JISのCSVファイルをUTF-8で保存する関数。
    
    Args:
        input_path (str): 入力CSVファイルパス
        output_path (str): 出力CSVファイルパス
    
    Raises:
        FileNotFoundError: 入力ファイルが存在しない場合
        UnicodeDecodeError: Shift-JISで解码できない文字の場合
        IOError: ファイル操作エラー
    """
    try:
        # 1. Shift-JISで読み込み（エラーレジストリーを設定）
        with open(input_path, 'r', encoding='shift_jis', errors='replace') as f_in:
            reader = csv.reader(f_in)
            rows = list(reader)
        
        # 2. UTF-8で書き込み
        with open(output_path, 'w', encoding='utf-8', newline='') as f_out:
            writer = csv.writer(f_out)
            writer.writerows(rows)
            
        print(f"成功: {input_path} → {output_path}")
        
    except FileNotFoundError:
        print(f"エラー: ファイル '{input_path}' が見つかりません。", file=sys.stderr)
        raise
        
    except UnicodeDecodeError as e:
        print(f"エラー: Shift-JIS解码エラー - {e}", file=sys.stderr)
        raise
        
    except IOError as e:
        print(f"エラー: ファイル操作エラー - {e}", file=sys.stderr)
        raise
        
    except Exception as e:
        print(f"予期外のエラー: {e}", file=sys.stderr)
        raise
