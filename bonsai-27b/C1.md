## 2. 基本的なコード例

### (A) データベース接続の設定

```python
from sqlalchemy import create_engine, Column, Integer, String, Float
from sqlalchemy.orm import declarative_base, sessionmaker

# SQLite を使用（他の DB も可能）
DATABASE_URL = "sqlite:///example.db"
engine = create_engine(DATABASE_URL, echo=True)  # echo=True で SQL が出力される

Base = declarative_base()
SessionLocal = sessionmaker(bind=engine)
```

### (B) モデルの定義

```python
class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(50), nullable=False)
    email = Column(String(120), unique=True, nullable=False)
    age = Column(Float, default=0.0)

class Post(Base):
    __tablename__ = "posts"

    id = Column(Integer, primary_key=True, index=True)
    title = Column(String(200), nullable=False)
    content = Column(String, nullable=True)
    author_id = Column(Integer, nullable=False)  # ForeignKey に相当する列

# 関係（ForeignKey）を明示的に定義するには、通常は relationship を使うが、
```

### (C) セッションの作成とデータ操作

```python
def get_session():
    """セッションオブジェクトを取得する"""
    session = SessionLocal()
    try:
        yield session
    finally:
        session.close()


# --- 挿入（Create） ---
def create_user(name: str, email: str, age: float):
    with get_session() as session:
        user = User(name=name, email=email, age=age)
        session.add(user)
        session.commit()
        print(f"ユーザー '{name}' が作成されました。ID: {user.id}")


# --- 読み取り（Read） ---
def get_all_users():
    with get_session() as session:
        users = session.query(User).all()
        for u in users:
            print(f"ID: {u.id}, Name: {u.name}, Email: {u.email}")


def get_user_by_id(user_id: int):
    with get_session() as session:
        user = session.query(User).filter(User.id == user_id).first()
        return user


# --- 更新（Update） ---
def update_user_age(user_id: int, new_age: float):
    with get_session() as session:
        user = session.query(User).filter(User.id == user_id).first()
        if user:
            user.age = new_age
            session.commit()
            print(f"ユーザー {user_id} の年齢を更新しました。")


# --- 削除（Delete） ---
def delete_user(user_id: int):
    with get_session() as session:
        user = session.query(User).filter(User.id == user_id).first()
        if user:
            session.delete(user)
            session.commit()
            print(f"ユーザー {user_id} が削除されました。")


# --- 検索（Filter） ---
def get_users_by_name_pattern(pattern: str):
    with get_session() as session:
        users = session.query(User).filter(
            User.name.like(f"%{pattern}%")
        ).all()
        for u in users:
            print(u.name)
```

### (D) 関係（ForeignKey）の例

```python
from sqlalchemy import ForeignKey

class Post(Base):
    __tablename__ = "posts"

    id = Column(Integer, primary_key=True, index=True)
    title = Column(String(200), nullable=False)
    content = Column(String, nullable=True)
    author_id = Column(Integer, ForeignKey("users.id"), nullable=False)


# 関係定義（relationship）
User.posts = relationship("Post", back_populates="author")
Post.author = relationship("User", back_populates="posts")

# 使用例：ユーザーに属する投稿を取得
def get_user_posts(user_id: int):
    with get_session() as session:
        user = session.query(User).filter(User.id == user_id).first()
        if user:
            for post in user.posts:
                print(f"  - {post.title}")
```
