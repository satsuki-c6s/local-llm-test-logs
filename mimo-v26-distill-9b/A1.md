```python
"""Shift-JIS CSVの読み込み・UTF-8書き出しユーティリティ。"""

import csv
import io
import os
import sys
import traceback


def read_sjis_csv(path: str) -> list[dict]:
    """Shift-JISエンコードのCSVを読み込み、UTF-8文字列のリストとして返す。

    Args:
        path: CSVファイルのパス

    Returns:
        UTF-8文字列の行リスト（辞書形式）

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: Shift-JISとしてデコードできない場合
        csv.Error: CSV構文エラー
        OSError: ファイルを開けない場合
    """
    if not os.path.exists(path):
        raise FileNotFoundError(
            f"CSVファイルが見つかりません: {path}"
        )

    with open(path, "rb") as f:
        data = f.read()

    try:
        text = data.decode("shift_jis")
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            f"Shift-JISとしてデコードできません: {e.reason} (offset={e.start})"
        ) from e

    try:
        return list(csv.DictReader(io.StringIO(text, encoding="utf-8")))
    except csv.Error as e:
        raise csv.Error(f"CSVの構文エラー: {e}") from e


def write_utf8_csv(path: str, rows: list[dict]) -> None:
    """UTF-8文字列のリストをUTF-8エンコードのCSVとして保存する。

    Args:
        path: 保存先のパス
        rows: 保存する行のリスト
    """
    with open(path, "w", encoding="utf-8", newline="\n") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0].keys()) if rows else [])
        writer.writeheader()
        writer.writerows(rows)


def convert_sjis_csv_to_utf8(
    source_path: str,
    dest_path: str | None = None,
    encoding: str = "shift_jis",
) -> str:
    """Shift-JIS CSVをUTF-8 CSVに変換する。

    Args:
        source_path: 入力CSVのパス
        dest_path: 出力CSVのパス（省略時はsource_pathと同じディレクトリに
            拡張子をutf-8.csvに変更）
        encoding: 入力ファイルのエンコード（デフォルト: shift_jis）

    Returns:
        生成されたUTF-8 CSVファイルのパス
    """
    if dest_path is None:
        base = os.path.splitext(source_path)[0]
        dest_path = base + ".utf-8.csv"

    try:
        rows = read_sjis_csv(source_path)
    except FileNotFoundError as e:
        print(f"エラー: {e}", file=sys.stderr)
        return None
    except (UnicodeDecodeError, csv.Error, OSError) as e:
        print(f"エラー: {e}", file=sys.stderr)
        traceback.print_exc()
        return None

    try:
        write_utf8_csv(dest_path, rows)
    except OSError as e:
        print(f"エラー: {e}", file=sys.stderr)
        traceback.print_exc()
        return None

    print(f"変換完了: {source_path} -> {dest_path} ({len(rows)}行)")
    return dest_path


if __name__ == "__main__":
    convert_sjis_csv_to_utf8(sys.argv[1] if len(sys.argv) > 1 else "input.csv")
```

**使い方:**

```python
convert_sjis_csv_to_utf8("input.csv")  # input.csv -> input.csv.utf-8.csv
```

**主なポイント:**

- **存在しないファイル**: `os.path.exists()` で事前チェックし、`FileNotFoundError` を明示的にメッセージ付きで送出
- **文字化け**: バイナリを先に読み取り、`shift_jis` でデコードしてからCSV解析。失敗時は `UnicodeDecodeError` を詳細情報付きで送出
- **CSV構文エラー**: `csv.Error` を個別に捕捉
- **出力**: `newline="\n"` を指定して行末改行を制御し、`utf-8` エンコードで保存
- **例外処理**: 変換全体を `try/except` で囲み、失敗時は `traceback.print_exc()` で詳細を出力しつつ、処理を継続可能にする設計
