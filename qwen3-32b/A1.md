### ✅ Pythonコード

```python
import csv
import os

def convert_csv_encoding(input_path, output_path):
    """
    Shift-JISでエンコードされたCSVファイルをUTF-8で保存し直す関数。
    
    Args:
        input_path (str): 入力CSVファイルのパス（Shift-JIS）
        output_path (str): 出力CSVファイルのパス（UTF-8）
    
    Returns:
        bool: 成功した場合はTrue、失敗した場合はFalse
    """
    try:
        # ファイルが存在するか確認
        if not os.path.exists(input_path):
            raise FileNotFoundError(f"ファイルが見つかりません: {input_path}")

        # Shift-JISで読み込み
        with open(input_path, 'r', encoding='shift_jis') as f:
            reader = csv.reader(f)
            data = [row for row in reader]

        # UTF-8で保存
        with open(output_path, 'w', encoding='utf-8', newline='') as f:
            writer = csv.writer(f)
            writer.writerows(data)

        print(f"変換成功: {input_path} → {output_path}")
        return True

    except FileNotFoundError as e:
        print(f"エラー: {e}")
        return False

    except UnicodeDecodeError as e:
        print(f"文字化けエラー（デコード失敗）: {e}")
        return False

    except Exception as e:
        print(f"予期しないエラー: {e}")
        return False

# 使用例
if __name__ == "__main__":
    input_file = "input.csv"      # Shift-JISでエンコードされたCSV
    output_file = "output_utf8.csv"  # UTF-8で保存するCSV

    result = convert_csv_encoding(input_file, output_file)
    if result:
        print("処理が正常に完了しました。")
    else:
        print("処理に失敗しました。")
```

---

### ✅ 補足説明

- `encoding='shift_jis'` で Shift-JIS から UTF-8 への変換を試みます。
- `newline=''` を指定することで、CSVの行末処理を OS に依存せず統一します。
- `UnicodeDecodeError` は文字化け（デコード失敗）の際に発生します。
- `FileNotFoundError` はファイルが存在しない場合に発生します。
