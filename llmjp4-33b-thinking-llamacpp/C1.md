## 2️⃣ モデル定義（テーブル作成）

`models.py` に **BaseModel** を継承したクラスを書きます。  
フィールドは **Field** クラスか、型ヒントだけで自動推論させても構いません。

```python
# models.py
from sakura_orm import BaseModel, Field, Index
from datetime import datetime
from typing import List

class User(BaseModel):
    """
    users テーブル
    """
    __tablename__ = "users"

    id: int = Field(primary_key=True)               # 自動インクリメント PK
    username: str = Field(max_length=50, unique=True)
    email: str = Field(max_length=120, nullable=False)
    created_at: datetime = Field(default_factory=datetime.utcnow)

    # インデックス例（検索高速化）
    __indexes__ = [
        Index(fields=["email"], name="idx_user_email")
    ]

    # One-to-Many の逆参照（User → Post）
    posts: List["Post"] = Field(relation="one_to_many", related_name="author")


class Post(BaseModel):
    """
    posts テーブル
    """
    __tablename__ = "posts"

    id: int = Field(primary_key=True)
    title: str = Field(max_length=200)
    body: str = Field()
    published_at: datetime = Field(nullable=True)

    # 外部キー（User.id）へのリレーション
    author_id: int = Field(foreign_key="users.id")
    author: User = Field(relation="many_to_one", related_name="posts")
```

### フィールドオプションまとめ

| オプション | 説明 |
|------------|------|
| `primary_key=True` | 主キー（自動採番） |
| `unique=True` | ユニーク制約 |
| `nullable=False` | NULL 不可 |
| `default=...` | 固定値デフォルト |
| `default_factory=callable` | 呼び出し可能オブジェクトで遅延評価（日時等） |
| `max_length=n` | VARCHAR(n) 相当 |
| `foreign_key="table.column"` | 外部キー制約 |
| `relation="one_to_many"` / `"many_to_one"` / `"many_to_many"` | リレーション定義 |
| `related_name="..."` | 逆参照属性名 |

## 3️⃣ データベース初期化

```python
# init_db.py
import asyncio
from sakura_orm import create_engine, run_sync
from settings import DATABASE_URL
from models import User, Post

# エンジン生成（同期/非同期どちらでも可）
engine = create_engine(DATABASE_URL, echo=True)   # echo=True で SQL ログ出力

def sync_main():
    # 同期コンテキストでテーブル作成
    run_sync(engine.create_all())
    print("✅ All tables created.")

if __name__ == "__main__":
    asyncio.run(run_sync(sync_main()))
```

- `engine.create_all()` が **全モデルのテーブルを一括作成**します。  
- 既存テーブルがある場合は **DROP** せずに **ALTER**（カラム追加）を試みます（ただし完全なマイグレーション機能は別ツール `sakura-migrate` 推奨）。

## 4️⃣ CRUD（Create / Read / Update / Delete）

以下は **同期 API** の例です。非同期版は `await` と `async with` に置き換えるだけです。

```python
# crud_demo.py
import asyncio
from sakura_orm import Session, select
from settings import DATABASE_URL
from models import User, Post

engine = create_engine(DATABASE_URL, echo=False)

# ---------- CREATE ----------
def create_user(username: str, email: str) -> User:
    with Session(engine) as sess:
        user = User(username=username, email=email)
        sess.add(user)            # INSERT 文が自動生成
        sess.commit()             # コミットすると PK が確定
        return user

# ---------- READ ----------
def get_user_by_id(user_id: int) -> User | None:
    with Session(engine) as sess:
        stmt = select(User).where(User.id == user_id)
        result = sess.execute(stmt).scalar_one_or_none()
        return result

def list_users(limit: int = 10, offset: int = 0):
    with Session(engine) as sess:
        stmt = select(User).order_by(User.created_at.desc()).limit(limit).offset(offset)
        return sess.scalars(stmt).all()

# ---------- UPDATE ----------
def update_email(user_id: int, new_email: str) -> bool:
    with Session(engine) as sess:
        stmt = (
            select(User)
            .where(User.id == user_id)
            .with_for_update()      # 行ロック（必要なら）
        )
        user = sess.execute(stmt).scalar_one()
        if user:
            user.email = new_email
            sess.commit()
            return True
        return False

# ---------- DELETE ----------
def delete_user(user_id: int) -> bool:
    with Session(engine) as sess:
        stmt = select(User).where(User.id == user_id)
        user = sess.execute(stmt).scalar_one()
        if user:
            sess.delete(user)
            sess.commit()
            return True
        return False

# -------------------- 実行例 --------------------
if __name__ == "__main__":
    # 初回だけテーブル作成
    from init_db import engine as _
    # サンプルデータ投入
    alice = create_user("alice", "alice@example.com")
    bob   = create_user("bob",   "bob@example.org")
    print(f"Created: {alice}, {bob}")

    # 検索
    fetched = get_user_by_id(alice.id)
    print("Fetched:", fetched)

    # 更新
    update_email(bob.id, "newbob@mail.com")
    print("Bob's new email:", get_user_by_id(bob.id).email)

    # 削除
    delete_user(alice.id)
    print("Alice deleted?", get_user_by_id(alice.id) is None)
```

### ポイント解説

| 操作 | キーワード |
|------|-------------|
| `Session(engine)` | コンテキストマネージャで自動コミット／ロールバック |
| `sess.add(obj)` | INSERT（新規レコード） |
| `sess.delete(obj)` | DELETE |
| `sess.commit()` | 明示的にトランザクション確定 |
| `select(...).where(...)` | ビルダー形式の SELECT |
| `scalar_one_or_none()` | 1 件取得（見つからなければ `None`） |
| `scalars(...).all()` | 複数件取得（モデルインスタンスのリスト） |

### 5‑2 集約・グルーピング

```python
from sakura_orm import func

stmt = (
    select(User.username, func.count(Post.id).label("post_cnt"))
    .join(Post, User.id == Post.author_id)
    .group_by(User.username)
    .having(func.count(Post.id) > 0)
    .order_by(func.count(Post.id).desc())
)

for row in engine.execute(stmt):
    print(row.username, row.post_cnt)
```

### 5‑3 ページング（Cursor based）

```python
def paginate_posts(page: int = 1, per_page: int = 20):
    offset = (page - 1) * per_page
    stmt = (
        select(Post)
        .order_by(Post.published_at.desc())
        .limit(per_page)
        .offset(offset)
    )
    return engine.scalars(stmt).all()
```

## 6️⃣ リレーション（One‑to‑Many・Many‑to‑Many）

### 6‑1 One‑to‑Many（User ↔ Post）

```python
# すでに models.py で relation 定義済みなので、アクセスはシンプル

# 例：ユーザーが持つ全投稿を取得
def user_posts(user_id: int):
    with Session(engine) as sess:
        user = sess.get(User, user_id)   # 主キーで即取得
        return user.posts                # ← lazy load（必要時に SELECT 発行）

# 例：投稿側から作者情報へアクセス
def post_author(post_id: int):
    with Session(engine) as sess:
        post = sess.get(Post, post_id)
        return post.author
```

### 6‑2 Many‑to‑Many（Tag ↔ Post）

```python
# models.py に中間テーブルとリレーションを追加

class Tag(BaseModel):
    __tablename__ = "tags"
    id: int = Field(primary_key=True)
    name: str = Field(unique=True)

# 中間テーブル（自動生成）
class PostTag(BaseModel):
    __tablename__ = "post_tags"
    post_id: int = Field(foreign_key="posts.id")
    tag_id: int = Field(foreign_key="tags.id")

# Post 側の many-to-many 定義
class Post(BaseModel):
    __tablename__ = "posts"
    # ... 既存カラム ...

    tags: List[Tag] = Field(
        relation="many_to_many",
        through="post_tags",
        related_name="posts"
    )
```

#### 使用例

```python
def add_tag_to_post(post_id: int, tag_name: str):
    with Session(engine) as sess:
        post = sess.get(Post, post_id)
        tag = sess.scalars(select(Tag).where(Tag.name == tag_name)).first()
        if not tag:
            tag = Tag(name=tag_name)
            sess.add(tag)
            sess.flush()           # tag.id が確定

        if tag not in post.tags:
            post.tags.append(tag)  # 中間テーブルに INSERT が走る
        sess.commit()
```

## 7️⃣ 非同期（async）での利用

Sakura‑ORM は **asyncio** 対応エンジン (`AsyncEngine`) を提供しています。  
非同期コードは `await` と `async with` が必須です。

```python
# async_demo.py
import asyncio
from sakura_orm import AsyncEngine, AsyncSession, select
from settings import DATABASE_URL
from models import User, Post

async_engine = AsyncEngine(DATABASE_URL, echo=False)

async def async_create_user(username: str, email: str) -> User:
    async with AsyncSession(async_engine) as sess:
        user = User(username=username, email=email)
        sess.add(user)
        await sess.commit()
        await sess.refresh(user)   # PK 取得
        return user

async def async_list_posts():
    async with AsyncSession(async_engine) as sess:
        stmt = select(Post).order_by(Post.published_at.desc())
        result = await sess.scalars(stmt)
        return result.all()

# 実行
if __name__ == "__main__":
    asyncio.run(async_create_user("charlie", "c@example.net"))
    posts = asyncio.run(async_list_posts())
    print(posts)
```

> **Tips**  
> - `await sess.refresh(obj)` は **INSERT 後に自動生成された PK や DEFAULT 値** をオブジェクトに反映させます。  
> - `AsyncEngine` は内部で `aiomysql`, `asyncpg`, `aiosqlite` 等のドライバを使用しますので、事前に `pip install aiosqlite asyncpg aiomysql` が必要です。

## 8️⃣ マイグレーション（簡易版）

Sakura‑ORM 本体は **DDL 自動生成** のみですが、公式拡張 `sakura-migrate` が **Alembic 互換** のマイグレーション管理を提供します。

```bash
pip install sakura-migrate
```

### 初期化

```bash
sakura-migrate init ./migrations
```

```bash
# 例：User テーブルに `is_active` カラムを追加
sakura-migrate revision -m "add is_active column to users"
```
