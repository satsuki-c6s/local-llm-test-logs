import csv
import sys
import os

def convert_csv_to_utf8(input_file, output_file=None):
    """
    Shift-JIS (可能日配配期先) CSV尔特文件读取，
    以UTF-8编码将内容写入新CSV文件。
    
    Args:
        input_file: 输入文件路径
        output_file: 输出文件路径 (如果未指定，默认为 input_file + '_utf8.csv')
    
    Returns:
        True:  conversed successfully
        False: 如果发生错误
    """
    if not output_file:
        output_file = input_file + '_utf8.csv'
    
    # 检查输入文件是否存在
    if not os.path.exists(input_file):
        print(f"错误: 输入文件 '{input_file}' 不存在.")
        return False
    
    # 打开Shift-JIS编码的CSV文件
    try:
        with open(input_file, 'r', encoding='shift-jis') as in_file:
            # 使用csv.reader读取CSV数据
            reader = csv.reader(in_file)
            rows = list(reader)
    except UnicodeDecodeError as e:
        print(f"错误: 无法以shift-jis编码读取 '{input_file}'.")
        print(f"  错误信息: {e}")
        return False
    except OSError as e:
        print(f"错误: 无法打开 '{input_file}'.")
        print(f"  错误信息: {e}")
        return False
    
    # 写入UTF-8编码的CSV文件
    try:
        with open(output_file, 'w', encoding='utf-8') as out_file:
            writer = csv.writer(out_file)
            writer.writerow(rows)
    except OSError as e:
        print(f"错误: 无法写入 '{output_file}'.")
        print(f"  错误信息: {e}")
        return False
    
    print(f"成功: '{input_file}' 已 conversed 到 '{output_file}'.")
    return True

# 主函数
if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("用法: python convert_csv.py <input_file> [output_file]")
        sys.exit(1)
    
    input_file = sys.argv[1]
    output_file = sys.argv[2] if len(sys.argv) > 2 else None
    
    convert_csv_to_utf8(input_file, output_file)
