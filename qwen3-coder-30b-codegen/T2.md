```html
<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>ブロック崩しゲーム</title>
  <style>
    body {
      margin: 0;
      padding: 0;
      display: flex;
      justify-content: center;
      align-items: center;
      height: 100vh;
      background-color: #000;
      color: white;
      font-family: Arial, sans-serif;
    }
    canvas {
      border: 1px solid #fff;
    }
    #message {
      position: absolute;
      text-align: center;
      font-size: 24px;
    }
  </style>
</head>
<body>
  <canvas id="gameCanvas" width="800" height="600"></canvas>
  <div id="message"></div>

  <script>
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');
    const messageDiv = document.getElementById('message');

    // ゲーム状態
    let gameRunning = true;
    let score = 0;

    // パドル
    const paddle = {
      width: 100,
      height: 15,
      x: canvas.width / 2 - 50,
      y: canvas.height - 30,
      speed: 8
    };

    // ボール
    const ball = {
      radius: 10,
      x: canvas.width / 2,
      y: canvas.height / 2,
      dx: 4,
      dy: -4,
      speed: 4
    };

    // ブロック
    const blockRows = 5;
    const blockColumns = 10;
    const blockWidth = 75;
    const blockHeight = 20;
    const blockPadding = 10;
    const blockOffsetTop = 60;
    const blockOffsetLeft = 30;

    let blocks = [];
    for (let r = 0; r < blockRows; r++) {
      blocks[r] = [];
      for (let c = 0; c < blockColumns; c++) {
        blocks[r][c] = { x: 0, y: 0, status: 1 };
      }
    }

    // キー入力
    const keys = {};
    document.addEventListener('keydown', (e) => {
      keys[e.key] = true;
    });
    document.addEventListener('keyup', (e) => {
      keys[e.key] = false;
    });

    // パドル移動
    function movePaddle() {
      if (keys['ArrowLeft'] && paddle.x > 0) {
        paddle.x -= paddle.speed;
      }
      if (keys['ArrowRight'] && paddle.x < canvas.width - paddle.width) {
        paddle.x += paddle.speed;
      }
    }

    // ボールの移動
    function moveBall() {
      ball.x += ball.dx;
      ball.y += ball.dy;

      // 壁との衝突
      if (ball.x + ball.radius > canvas.width || ball.x - ball.radius < 0) {
        ball.dx = -ball.dx;
      }

      // 天井との衝突
      if (ball.y - ball.radius < 0) {
        ball.dy = -ball.dy;
      }

      // パドルとの衝突
      if (
        ball.y + ball.radius > paddle.y &&
        ball.x > paddle.x &&
        ball.x < paddle.x + paddle.width
      ) {
        ball.dy = -Math.abs(ball.dy);
        // パドルの中心からの距離に応じて角度を変える
        const hitPosition = (ball.x - paddle.x) / paddle.width;
        ball.dx = (hitPosition - 0.5) * 8;
      }

      // ブロックとの衝突
      for (let r = 0; r < blockRows; r++) {
        for (let c = 0; c < blockColumns; c++) {
          const block = blocks[r][c];
          if (block.status === 1) {
            if (
              ball.x > block.x &&
              ball.x < block.x + blockWidth &&
              ball.y > block.y &&
              ball.y < block.y + blockHeight
            ) {
              ball.dy = -ball.dy;
              block.status = 0;
              score += 10;
            }
          }
        }
      }

      // ゲームオーバー
      if (ball.y + ball.radius > canvas.height) {
        gameRunning = false;
        messageDiv.textContent = 'GAME OVER';
        messageDiv.style.display = 'block';
      }

      // クリア
      let allCleared = true;
      for (let r = 0; r < blockRows; r++) {
        for (let c = 0; c < blockColumns; c++) {
          if (blocks[r][c].status === 1) {
            allCleared = false;
          }
        }
      }

      if (allCleared) {
        gameRunning = false;
        messageDiv.textContent = 'YOU WIN!';
        messageDiv.style.display = 'block';
      }
    }

    // 描画
    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // パドル描画
      ctx.fillStyle = '#fff';
      ctx.fillRect(paddle.x, paddle.y, paddle.width, paddle.height);

      // ボール描画
      ctx.beginPath();
      ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
      ctx.fillStyle = '#fff';
      ctx.fill();
      ctx.closePath();

      // ブロック描画
      for (let r = 0; r < blockRows; r++) {
        for (let c = 0; c < blockColumns; c++) {
          const block = blocks[r][c];
          if (block.status === 1) {
            block.x = c * (blockWidth + blockPadding) + blockOffsetLeft;
            block.y = r * (blockHeight + blockPadding) + blockOffsetTop;
            ctx.fillStyle = '#ff0000';
            ctx.fillRect(block.x, block.y, blockWidth, blockHeight);
          }
        }
      }

      // スコア表示
      ctx.font = '16px Arial';
      ctx.fillText(`Score: ${score}`, 10, 20);
    }

    // ゲームループ
    function gameLoop() {
      if (gameRunning) {
        movePaddle();
        moveBall();
        draw();
      }
      requestAnimationFrame(gameLoop);
    }

    // 初期化
    messageDiv.style.display = 'none';
    gameLoop();
  </script>
</body>
</html>
```
