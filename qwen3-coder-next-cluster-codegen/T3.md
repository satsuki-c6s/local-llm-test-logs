````
以下は、外部ライブラリやCDNを使わず、単一のHTMLファイルだけで動作する地球・太陽・月の軌道シミュレーターです。要件をすべて満たすように実装しています。

```html
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>地球・太陽・月の軌道シミュレーター</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            background-color: #000;
            color: white;
            font-family: sans-serif;
            display: flex;
            flex-direction: column;
            align-items: center;
            justify-content: center;
            height: 100vh;
        }
        h1 {
            margin: 10px 0;

        }
        #canvas-container {
            position: relative;

        }

</head>
<body>
    <h1>地球・太陽・月の軌道シミュレーター</h1>
    
    <div id="canvas-container">
        <canvas id="orbitCanvas" width="800" height="600"></canvas>
    </div>

    <div id="controls">
        <button id="toggleBtn">一時停止</button>
        <div class="speed-buttons">
            <button class="speed-btn active" data-speed="1">遅い</button>
            <button class="speed-btn active" data-speed="2">普通</button>
            <button class="speed-btn active" data-speed="4">速い</button>
        </div>
    </div>
    
    <div class="legend">
        <div class="legend-item">
            <div class="dot" style="background-color: #FFD700;"></div>
            <span>太陽</span>
        </div>
        <div class="legend-item">
            <div class="dot" style="background-color: #1E90FF;"></div>
            <span>地球</span>
        </div>
        <div class="legend-item">
            <div class="dot" style="background-color: #CCCCCC;"></div>
            <span>月</span>
        </div>
    </div>

    <script>
        // Canvasの設定
        const canvas = document.getElementById('orbitCanvas');
        const ctx = canvas.getContext('2d');
        const width = canvas.width;
        const height = canvas.height;
        const centerX = width / 2;
        const centerY = height / 2;

        // 天体の設定（見やすいように実際の縮尺は調整）
        const sunRadius = 30;
        const earthRadius = 12;
        const moonRadius = 5;
        const earthOrbitRadius = 150; // 太陽から地球までの距離
        const moonOrbitRadius = 30;   // 地球から月までの距離

        // 公転周期比：地球1周の間に月が約12周する
        // 地球の公転角速度を1とすると、月の公転角速度は12
        const earthSpeed = 0.015; // 基本速度（遅い設定）
        const moonSpeedRatio = 12; // 月の周期は地球の1/12（12回公転）

        // 状態管理
        let isPaused = false;
        let speedMultiplier = 1;
        let earthAngle = 0;
        let moonAngle = 0;

        // 太陽の光のエフェクト用
        let glowAlpha = 0.3;

        // 描画関数
        function draw() {
            // キャンバスをクリア
            ctx.fillStyle = 'black';
            ctx.fillRect(0, 0, width, height);

            // 太陽の描画
            drawSun();

            // 地球の軌道線を描画
            ctx.beginPath();
            ctx.strokeStyle = 'rgba(30, 144, 255, 0.3)';
            ctx.lineWidth = 1;
            ctx.arc(centerX, centerY, earthOrbitRadius, 0, Math.PI * 2);
            ctx.stroke();

            // 地球の位置を計算
            const earthX = centerX + Math.cos(earthAngle) * earthOrbitRadius;
            const earthY = centerY + Math.sin(earthAngle) * earthOrbitRadius;

            // 地球の描画
            drawEarth(earthX, earthY);

            // 月の軌道線を描画（地球中心）
            ctx.beginPath();
            ctx.strokeStyle = 'rgba(200, 200, 200, 0.2)';
            ctx.lineWidth = 1;
            ctx.setLineDash([3, 3]); // 点線
            ctx.arc(earthX, earthY, moonOrbitRadius, 0, Math.PI * 2);
            ctx.stroke();
            ctx.setLineDash([]); // 点線をリセット

            // 月の位置を計算
            const moonX = earthX + Math.cos(moonAngle) * moonOrbitRadius;
            const moonY = earthY + Math.sin(moonAngle) * moonOrbitRadius;

            // 月の描画
            drawMoon(moonX, moonY);

            // ラベルを描画
            drawLabels(earthX, earthY, moonX, moonY);

            // 太陽の光のエフェクトを更新
            glowAlpha = 0.3 + Math.sin(Date.now() / 500) * 0.1;
        }

        function drawSun() {
            // 太陽の本体
            ctx.beginPath();
            ctx.fillStyle = '#FFD700'; // 金黄色
            ctx.arc(centerX, centerY, sunRadius, 0, Math.PI * 2);
            ctx.fill();

            // 太陽の光（エフェクト）
            ctx.beginPath();
            ctx.fillStyle = `rgba(255, 215, 0, ${glowAlpha})`;
            ctx.arc(centerX, centerY, sunRadius * 1.5, 0, Math.PI * 2);
            ctx.fill();
            
            // 太陽のラベル
            ctx.fillStyle = 'white';
            ctx.font = 'bold 16px sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('太陽', centerX, centerY);
        }

        function drawEarth(x, y) {
            // 地球の本体
            ctx.beginPath();
            ctx.fillStyle = '#1E90FF'; // ドッジボールブルー
            ctx.arc(x, y, earthRadius, 0, Math.PI * 2);
            ctx.fill();

            // 地球のラベル
            ctx.fillStyle = 'white';
            ctx.font = 'bold 14px sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('地球', x, y - earthRadius - 10);
        }

        function drawMoon(x, y) {
            // 月の本体
            ctx.beginPath();
            ctx.fillStyle = '#CCCCCC'; // 灰色
            ctx.arc(x, y, moonRadius, 0, Math.PI * 2);
            ctx.fill();

            // 月のラベル
            ctx.fillStyle = 'white';
            ctx.font = 'bold 12px sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('月', x, y - moonRadius - 8);
        }

        function drawLabels(earthX, earthY, moonX, moonY) {
            // 軌道の補助線（オプション）
            ctx.beginPath();
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.1)';
            ctx.lineWidth = 1;
            // 地球と太陽を結ぶ線
            ctx.moveTo(centerX, centerY);
            ctx.lineTo(earthX, earthY);
            // 月と地球を結ぶ線
            ctx.moveTo(earthX, earthY);
            ctx.lineTo(moonX, moonY);
            ctx.stroke();
        }

        // アニメーションループ
        function update() {
            if (!isPaused) {
                // 角度を更新
                earthAngle += earthSpeed * speedMultiplier;
                moonAngle += earthSpeed * moonSpeedRatio * speedMultiplier;
            }
            
            draw();
            requestAnimationFrame(update);
        }

        // ボタンイベント
        const toggleBtn = document.getElementById('toggleBtn');
        toggleBtn.addEventListener('click', () => {
            isPaused = !isPaused;
            toggleBtn.textContent = isPaused ? '再開' : '一時停止';
            toggleBtn.style.backgroundColor = isPaused ? '#d9534f' : '#444';
        });

        // 速度調整ボタン
        const speedBtns = document.querySelectorAll('.speed-btn');
        speedBtns.forEach(btn => {
            btn.addEventListener('click', () => {
                // すべてのボタンのスタイルをリセット
                speedBtns.forEach(b => {
                    b.classList.remove('active');
                    b.style.backgroundColor = '#444';
                });
                
                // クリックされたボタンをアクティブに
                btn.classList.add('active');
                btn.style.backgroundColor = '#007bff';
                
                // 速度設定
                speedMultiplier = parseInt(btn.dataset.speed);
            });
        });

        // 初期化
        update();
    </script>
</body>
</html>
````
