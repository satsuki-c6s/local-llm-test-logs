## プロンプト

```
ブラウザで動くToDoアプリを、外部ライブラリやCDNを使わず、単一のHTMLファイル1つだけで作ってください。要件: (1) テキスト入力と追加ボタンでタスクを追加できる (2) 各タスクはチェックで完了/未完了を切り替えられる (3) 各タスクは削除できる (4) 各タスクはダブルクリックで内容を編集できる (5) 「すべて/未完了/完了」の3つのフィルタ切り替えがある (6) 未完了タスクの件数が表示される (7) localStorageに保存され、リロードしても消えない。コードはHTMLファイル全文を1つのコードブロックで出力してください。
```

## 応答全文 (そのまま)

以下に、指定された要件を満たす単一のHTMLファイルのコードを提供します。

```html
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>ToDo App</title>
    <style>
        body { font-family: Arial, sans-serif; margin: 20px; }
        #task-list { list-style-type: none; padding: 0; }
        #task-list li { display: flex; align-items: center; margin-bottom: 5px; }
        #task-list li.completed label { text-decoration: line-through; color: gray; }
        .controls { margin-bottom: 10px; }
        .filters button { margin-right: 5px; }
    </style>
</head>
<body>
    <h1>ToDo App</h1>
    <div class="controls">
        <input type="text" id="task-input" placeholder="新しいタスクを追加...">
        <button onclick="addTask()">追加</button>
        <span id="remaining-count">0 件の未完了タスク</span>
    </div>
    <div class="filters">
        <button onclick="filterTasks('all')">すべて</button>
        <button onclick="filterTasks('active')">未完了</button>
        <button onclick="filterTasks('completed')">完了</button>
    </div>
    <ul id="task-list"></ul>

    <script>
        const taskList = document.getElementById('task-list');
        const taskInput = document.getElementById('task-input');
        const remainingCount = document.getElementById('remaining-count');

        let tasks = JSON.parse(localStorage.getItem('tasks')) || [];

        function renderTasks() {
            taskList.innerHTML = '';
            tasks.forEach((task, index) => {
                const li = document.createElement('li');
                li.className = task.completed ? 'completed' : '';

                const checkbox = document.createElement('input');
                checkbox.type = 'checkbox';
                checkbox.checked = task.completed;
                checkbox.onchange = () => toggleTaskCompletion(index);

                const label = document.createElement('label');
                label.textContent = task.text;
                label.ondblclick = () => editTask(index, label);

                const deleteButton = document.createElement('button');
                deleteButton.textContent = '削除';
                deleteButton.onclick = () => deleteTask(index);

                li.appendChild(checkbox);
                li.appendChild(label);
                li.appendChild(deleteButton);
                taskList.appendChild(li);
            });
            updateRemainingCount();
        }

        function addTask() {
            const text = taskInput.value.trim();
            if (text) {
                tasks.push({ text, completed: false });
                taskInput.value = '';
                saveTasks();
                renderTasks();
            }
        }

        function toggleTaskCompletion(index) {
            tasks[index].completed = !tasks[index].completed;
            saveTasks();
            renderTasks();
        }

        function deleteTask(index) {
            tasks.splice(index, 1);
            saveTasks();
            renderTasks();
        }

        function editTask(index, label) {
            const input = document.createElement('input');
            input.type = 'text';
            input.value = tasks[index].text;
            input.onblur = () => {
                tasks[index].text = input.value.trim() || tasks[index].text;
                saveTasks();
                renderTasks();
            };
            label.parentNode.replaceChild(input, label);
            input.focus();
        }

        function filterTasks(filter) {
            taskList.querySelectorAll('li').forEach(li => {
                const isCompleted = li.classList.contains('completed');
                if (filter === 'all' || 
                    (filter === 'active' && !isCompleted) || 
                    (filter === 'completed' && isCompleted)) {
                    li.style.display = '';
                } else {
                    li.style.display = 'none';
                }
            });
        }

        function updateRemainingCount() {
            const count = tasks.filter(task => !task.completed).length;
            remainingCount.textContent = `${count} 件の未完了タスク`;
        }

        function saveTasks() {
            localStorage.setItem('tasks', JSON.stringify(tasks));
        }

        renderTasks();
    </script>
</body>
</html>
```

このコードは、指定された要件を満たすToDoアプリケーションを作成します。HTML, CSS, JavaScriptのみを使用しており、外部ライブラリやCDNは使用していません。タスクの追加、完了/未完了の切り替え、削除、編集、フィルタリング、未完了タスク数の表示、およびlocalStorageへの保存が可能です。
