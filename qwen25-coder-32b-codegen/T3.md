## 応答全文 (そのまま)

以下に、指定された要件に基づいた単一のHTMLファイルのソースコードを提供します。

```html
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>Planetary Orbit Simulator</title>
<style>
  body { font-family: Arial, sans-serif; text-align: center; }
  canvas { display: block; margin: auto; background-color: #000; }
  button { margin: 5px; padding: 10px 20px; font-size: 16px; cursor: pointer; }
</style>
</head>
<body>

<h1>Planetary Orbit Simulator</h1>
<canvas id="simulator" width="800" height="600"></canvas>
<div>
  <button onclick="toggleAnimation()">一時停止/再開</button>
  <button onclick="setSpeed(0.5)">遅い</button>
  <button onclick="setSpeed(1)">普通</button>
  <button onclick="setSpeed(2)">速い</button>
</div>

<script>
const canvas = document.getElementById('simulator');
const ctx = canvas.getContext('2d');

let running = true;
let speed = 1;

function toggleAnimation() {
  running = !running;
}

function setSpeed(newSpeed) {
  speed = newSpeed;
}

function drawOrbit(x, y, radius, color) {
  ctx.beginPath();
  ctx.arc(x, y, radius, 0, Math.PI * 2);
  ctx.strokeStyle = color;
  ctx.stroke();
}

function drawPlanet(x, y, size, color, label) {
  ctx.fillStyle = color;
  ctx.beginPath();
  ctx.arc(x, y, size, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = '#fff';
  ctx.font = '14px Arial';
  ctx.fillText(label, x + size + 5, y + 5);
}

function animate() {
  if (!running) return requestAnimationFrame(animate);

  const centerX = canvas.width / 2;
  const centerY = canvas.height / 2;

  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // Draw orbits
  drawOrbit(centerX, centerY, 150, '#888'); // Earth's orbit
  drawOrbit(centerX + 150 * Math.cos(speed * Date.now() / 1000), centerY + 150 * Math.sin(speed * Date.now() / 1000), 30, '#aaa'); // Moon's orbit

  // Draw Sun
  drawPlanet(centerX, centerY, 20, '#ffcc00', 'Sun');

  // Calculate Earth position
  const earthAngle = speed * Date.now() / 1000;
  const earthX = centerX + 150 * Math.cos(earthAngle);
  const earthY = centerY + 150 * Math.sin(earthAngle);

  // Draw Earth
  drawPlanet(earthX, earthY, 12, '#0044ff', 'Earth');

  // Calculate Moon position
  const moonAngle = speed * Date.now() / 1000 * 12;
  const moonX = earthX + 30 * Math.cos(moonAngle);
  const moonY = earthY + 30 * Math.sin(moonAngle);

  // Draw Moon
  drawPlanet(moonX, moonY, 6, '#cccccc', 'Moon');

  requestAnimationFrame(animate);
}

animate();
</script>

</body>
</html>
```

このコードは、単一のHTMLファイルで地球・太陽・月の軌道シミュレーターを作成します。canvasを使用して描画し、各天体の軌道線とラベルを表示します。また、一時停止/再開ボタンと速度調整機能も含まれています。
