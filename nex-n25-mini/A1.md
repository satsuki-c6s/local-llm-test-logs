## 出力

以下は、Shift-JIS の CSV ファイルを読み込んで UTF-8 で保存し直す関数です。  
`utf-8-sig` で保存することで、Excel などで開いたときに文字化けしにくくなります。

```python
import csv
import os
from pathlib import Path
from typing import Optional


def shift_jis_csv_to_utf8(
    input_path: str | os.PathLike[str],
    output_path: str | os.PathLike[str] | None = None,
    encoding: str = "cp932",
    errors: str = "strict",
    newline: str = "",
) -> Path:
    """
    Shift-JIS の CSV ファイルを読み込んで UTF-8 で保存し直す。

    Args:
        input_path:
            入力 CSV ファイルのパス
        output_path:
            出力 CSV ファイルのパス。
            None の場合は入力ファイル名に "_utf8" を付けたものになる。
        encoding:
            入力ファイルの文字コード。
            Shift-JIS 互換なら "cp932" を推奨。
        errors:
            デコード時のエラー処理。
            "strict" は文字化けや不正バイトでエラーにする。
        newline:
            csv モジュールの newline 引数。
            "" を推奨。

    Returns:
        出力ファイルのパス

    Raises:
        FileNotFoundError:
            入力ファイルが存在しない場合
        UnicodeDecodeError:
            Shift-JIS としてデコードできない場合
        OSError:
            出力ファイルの作成・書き込みに失敗した場合
    """
    input_path = Path(input_path)
    output_path = Path(output_path) if output_path is not None else input_path.with_name(
        f"{input_path.stem}_utf8{input_path.suffix}"
    )

    if not input_path.exists():
        raise FileNotFoundError(f"入力ファイルが存在しません: {input_path}")

    if not input_path.is_file():
        raise IsADirectoryError(f"入力パスがファイルではありません: {input_path}")

    with input_path.open("r", encoding=encoding, errors=errors, newline=newline) as f_in:
        reader = csv.reader(f_in)

        with output_path.open("w", encoding="utf-8-sig", newline=newline) as f_out:
            writer = csv.writer(f_out)

            for row in reader:
                writer.writerow(row)

    return output_path
```

使用例:

```python
try:
    output = shift_jis_csv_to_utf8("input.csv")
    print(f"変換完了: {output}")

except FileNotFoundError as e:
    print(f"ファイルが見つかりません: {e}")

except UnicodeDecodeError as e:
    print(f"文字コードのデコードに失敗しました: {e}")

except OSError as e:
    print(f"ファイルの読み書きに失敗しました: {e}")
```

注意点:

- Shift-JIS 互換の文字コードは Python では `cp932` がおすすめです。
- `errors="strict"` にしているため、Shift-JIS として不正なバイト列が含まれていたら `UnicodeDecodeError` になります。
- 出力は `utf-8-sig` にしています。BOM 付き UTF-8 です。  
  BOM 不要の場合は `encoding="utf-8"` に変更してください。
