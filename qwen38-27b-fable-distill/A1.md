```python
import csv
from pathlib import Path


def convert_csv_encoding(
    input_path: str | Path,
    output_path: str | Path,
    delimiter: str = ",",
) -> None:
    """Shift-JISのCSVを読み込み、UTF-8(UTF-8 BOM付き)で保存する。

    Excelで開けるようにUTF-8 BOM付きで書き出します。
    BOM不要なら encoding="utf-8" に変更してください。
    """
    input_path = Path(input_path)
    output_path = Path(output_path)

    if not input_path.exists():
        raise FileNotFoundError(f"ファイルが見つかりません: {input_path}")

    rows = []
    try:
        with input_path.open("r", encoding="shift_jis", newline="") as f:
            reader = csv.reader(f, delimiter=delimiter)
            rows = list(reader)
    except UnicodeDecodeError as e:
        raise ValueError(
            f"Shift-JISとしてデコードできませんでした(文字化けの可能性): {input_path}"
        ) from e

    with output_path.open("w", encoding="utf-8-sig", newline="") as f:
        writer = csv.writer(f, delimiter=delimiter)
        writer.writerows(rows)


if __name__ == "__main__":
    try:
        convert_csv_encoding("data.csv", "data_utf8.csv")
    except FileNotFoundError as e:
        print(f"エラー: {e}")
    except ValueError as e:
        print(f"エラー: {e}")
```

ポイント:

- **`newline=""`**: CSVの改行処理をcsvモジュールに任せるため必須です。
- **`utf-8-sig`**: BOM付きUTF-8で、Excelが文字化けせずに開けます。BOM不要なら `"utf-8"` に変更。
- **エラー処理**: ファイル不存在は `FileNotFoundError`、デコード失敗は `ValueError` を送出。呼び出し側でキャッチして表示しています。
- 読み込み時に全行をメモリに読み込む実装です。巨大ファイルの場合は行ごとにストリーミング書き出しする形に書き換える必要があります。
