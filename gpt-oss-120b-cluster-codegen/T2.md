## 応答全文 (そのまま)

````
以下のコードを **1 つの HTML ファイル** として保存すれば、ブラウザだけで動くブロック崩し（Breakout）ゲームがプレイできます。  
要件をすべて満たすように実装しています。

```html
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>ブロック崩しゲーム</title>
<style>
  body { margin:0; background:#111; color:#fff; font-family:sans-serif; }
  canvas { background:#000; display:block; margin:20px auto; }
  #info { text-align:center; margin-top:-10px; }
  #msg { position:fixed; top:0; left:0; width:100%; height:100%;
          display:flex; align-items:center; justify-content:center;
          background:rgba(0,0,0,0.8); color:#fff; font-size:2rem; }
</style>
</head>
<body>

<canvas id="gameCanvas" width="480" height="320"></canvas>
<div id="info">
  <span id="score">スコア: 0</span> |
  <span id="lives">残り: 3</span>
</div>
<div id="msg" style="display:none;"></div>

<script>
(() => {
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');

  // ゲーム設定
  const paddle = { width: 80, height: 10, x: 0, speed: 6 };
  const ball = { radius: 5, x: canvas.width/2, y: canvas.height-30, dx: 2, dy: -2 };
  const brick = { rowCount: 5, colCount: 8, width: 50, height: 20, padding: 5, offsetTop: 30, offsetLeft: 30 };
  let score = 0;
  let lives = 3;
  let rightPressed = false, leftPressed = false;
  let bricks = [];

  // ブロック生成
  function initBricks() {
    bricks = [];
    for(let r=0; r<brick.rowCount; r++) {
      bricks[r] = [];
      for(let c=0; c<brick.colCount; c++) {
        const x = (c*(brick.width+brick.padding))+brick.offsetLeft;
        const y = (r*(brick.width+brick.padding))+brick.offsetTop;
        bricks[r][c] = { x, y, status: 1 };
      }
    }
  }
  initBricks();

  // 描画
  function drawPaddle() {
    ctx.beginPath();
    ctx.rect(paddle.x, canvas.height-paddle.height-10, paddle.width, paddle.height);
    ctx.fillStyle = "#0095DD";
    ctx.fill();
    ctx.closePath();
  }

  function drawBall() {
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI*2);
    ctx.fillStyle = "#FF4500";
    ctx.fill();
    ctx.closePath();
  }

  function drawBricks() {
    for(let r=0; r<brick.rowCount; r++) {
      for(let c=0; c<brick.colCount; c++) {
        const b = bricks[r][c];
        if(b.status===1) {
          const bx = b.x;
          const by = b.y;
          ctx.beginPath();
          ctx.rect(bx, by, brick.width, brick.height);
          ctx.fillStyle = "#"+((r*50)%255).toString(16)+(c*30).toString(16)+"00";
          ctx.fill();
          ctx.closePath();
        }
      }
    }
  }

  function drawScore() {
    document.getElementById('score').textContent = `スコア: ${score}`;
  }

  function drawLives() {
    document.getElementById('lives').textContent = `残り: ${lives}`;
  }

  // キー入力
  document.addEventListener('keydown', e => {
    if(e.key === "Right" || e.key === "ArrowRight") rightPressed = true;
    else if(e.key === "Left" || e.key === "ArrowLeft") leftPressed = true;
  });
  document.addEventListener('keyup', e => {
    if(e.key === "Right" || e.key === "ArrowRight") rightPressed = false;
    else if(e.key === "Left" || e.key === "ArrowLeft") leftPressed = false;
  });

  // メイン描画ループ
  function draw() {
    ctx.clearRect(0,0,canvas.width,canvas.height);
    drawBricks();
    drawBall();
    drawPaddle();
    drawScore();
    drawLives();

    // ボールとブロックの衝突判定
    for(let r=0; r<brick.rowCount; r++) {
      for(let c=0; c<brick.colCount; c++) {
        const b = bricks[r][c];
        if(b.status===1) {
          if(ball.x > b.x && ball.x < b.x+brick.width && ball.y > b.y && ball.y < b.y+brick.height) {
            ball.dy = -ball.dy;
            b.status = 0;
            score++;
            if(score === brick.rowCount*brick.colCount) {
              showMessage('クリア！');
            }
          }
        }
      }
    }

    // 壁との衝突
    if(ball.x + ball.dx > canvas.width-ball.radius || ball.x + ball.dx < ball.radius) {
      ball.dx = -ball.dx;
    }
    if(ball.y + ball.dy < ball.radius) {
      ball.dy = -ball.dy;
    } else if(ball.y + ball.dy > canvas.height-ball.radius) {
      // パドルとの衝突判定
      if(ball.x > paddle.x && ball.x < paddle.x + paddle.width) {
        // 当たった位置で反射角を変える
        const hitPos = (ball.x - (paddle.x + paddle.width/2)) / (paddle.width/2);
        const angle = hitPos * Math.PI/3; // -60°〜+60°
        const speed = Math.sqrt(ball.dx*ball.dx + ball.dy*ball.dy);
        ball.dx = speed * Math.sin(angle);
        ball.dy = -Math.abs(speed * Math.cos(angle));
      } else {
        // ミス
        lives--;
        if(lives===0) {
          showMessage('ゲームオーバー');
        } else {
          // ボールとパドルをリセット
          ball.x = canvas.width/2;
          ball.y = canvas.height-30;
          ball.dx = 2;
          ball.dy = -2;
          paddle.x = (canvas.width-paddle.width)/2;
        }
      }
    }

    // パドル移動
    if(rightPressed && paddle.x < canvas.width-paddle.width) paddle.x += paddle.speed;
    else if(leftPressed && paddle.x > 0) paddle.x -= paddle.speed;

    requestAnimationFrame(draw);
  }

  function showMessage(txt) {
    const msgDiv = document.getElementById('msg');
    msgDiv.textContent = txt;
    msgDiv.style.display = 'flex';
    // 再スタートボタン
    const btn = document.createElement('button');
    btn.textContent = 'もう一度';
    btn.style.marginTop = '20px';
    btn.style.padding = '10px 20px';
    btn.style.fontSize = '1rem';
    btn.onclick = () => {
      // リセット
      score = 0;
      lives = 3;
      ball.x = canvas.width/2;
      ball.y = canvas.height-30;
      ball.dx = 2;
      ball.dy = -2;
      paddle.x = (canvas.width-paddle.width)/2;
      initBricks();
      msgDiv.style.display = 'none';
    };
    msgDiv.appendChild(document.createElement('br'));
    msgDiv.appendChild(btn);
  }

  // 初期化
  paddle.x = (canvas.width - paddle.width) / 2;
  draw();
})();
</script>
</body>
</html>
```

### 使い方
1. 上記コードを `breakout.html` など好きな名前で保存してください。  
2. そのファイルをブラウザで開くだけでゲームが起動します。  

- **左・右矢印キー**でパドルを操作します。  
- ボールがブロックに当たるとブロックが消え、スコアが 1 加算されます。  
- ボールが画面下部へ落ちるとライフが 1 減り、0 になると「ゲームオーバー」表示が出ます。  
- 全ブロックを破壊すると「クリア」メッセージが表示されます。  

すべて **1 つの HTML ファイル** だけで完結し、外部ライブラリや CDN は使用していません。ぜひお楽しみください！
````
