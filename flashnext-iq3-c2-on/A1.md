## 出力

以下は、Shift-JISのCSVファイルをUTF-8で書き直す関数です。  
CSVの内容をパースして書き換えるのではなく、文字コードだけを変換して、元の行末や引用符をできるだけ保持します。

```python
from pathlib import Path
from typing import Union


class CsvConversionError(Exception):
    """CSV変換失敗を表す基底例外"""


class CsvDecodeError(CsvConversionError):
    """Shift-JISとしてデコードできなかったことを表す例外"""


def convert_shiftjis_csv_to_utf8(
    src_path: Union[str, Path],
    dst_path: Union[str, Path],
    *,
    src_encoding: str = "shift_jis",
) -> None:
    """
    Shift-JISのCSVファイルをUTF-8で書き直す。

    Args:
        src_path: 入力CSVファイルパス
        dst_path: 出力CSVファイルパス
        src_encoding: 入力ファイルの文字コード。既定は "shift_jis"

    Raises:
        FileNotFoundError: 入力ファイルが存在しない場合
        CsvDecodeError: Shift-JISとしてデコードできなかった場合
        CsvConversionError: 読み書きその他の変換処理に失敗した場合
    """
    src = Path(src_path)
    dst = Path(dst_path)

    # 入力ファイル読み込み
    try:
        with src.open(
            "r",
            encoding=src_encoding,
            errors="strict",
            newline="",
        ) as fin:
            text = fin.read()

    except FileNotFoundError as e:
        raise FileNotFoundError(f"入力ファイルが見つかりません: {src}") from e

    except UnicodeDecodeError as e:
        raise CsvDecodeError(
            f"{src_encoding}としてデコードできませんでした: {src}"
        ) from e

    except LookupError as e:
        raise CsvConversionError(
            f"不正なエンコーディング名です: {src_encoding}"
        ) from e

    except OSError as e:
        raise CsvConversionError(
            f"入力ファイルの読み込みに失敗しました: {src}"
        ) from e

    # UTF-8で書き込み
    try:
        dst.parent.mkdir(parents=True, exist_ok=True)

        with dst.open(
            "w",
            encoding="utf-8",
            errors="strict",
            newline="",
        ) as fout:
            fout.write(text)

    except UnicodeEncodeError as e:
        raise CsvConversionError(
            f"UTF-8としてエンコードできませんでした: {dst}"
        ) from e

    except OSError as e:
        raise CsvConversionError(
            f"出力ファイルの書き込みに失敗しました: {dst}"
        ) from e
```

使い方例です。

```python
try:
    convert_shiftjis_csv_to_utf8("input.csv", "output.csv")

except FileNotFoundError as e:
    print("ファイルがありません:", e)

except CsvDecodeError as e:
    print("文字化け/デコード失敗:", e)

except CsvConversionError as e:
    print("変換失敗:", e)
```

補足です。

- `newline=""` はCSVの行末を勝手に変換しないために重要です。
- Windowsの日本語CSVでは、`shift_jis` ではなく `cp932` の方が成功することがあります。

```python
convert_shiftjis_csv_to_utf8(
    "input.csv",
    "output.csv",
    src_encoding="cp932",
)
```

- Excel用にBOM付きUTF-8にしたい場合は、書き込み側を `utf-8-sig` にしてください。

```python
with dst.open("w", encoding="utf-8-sig", errors="strict", newline="") as fout:
    fout.write(text)
```
