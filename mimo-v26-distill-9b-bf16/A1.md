```python
"""Shift-JIS CSVの読み込み・UTF-8書き出し。"""

import csv
import os
from pathlib import Path


def convert_sjis_csv_to_utf8(
    source: str | Path,
    destination: str | Path | None = None,
    encoding: str = "shift_jis",
    output_encoding: str = "utf-8",
) -> Path:
    """Shift-JISのCSVを読み込み、UTF-8で保存し直す。

    存在しないファイル・デコード失敗・エンコード失敗をすべて
    例外として返す（失敗時は部分書き出しを避けるため）。

    Args:
        source: 読み込むCSVファイルのパス。
        destination: 書き出すCSVファイルのパス。省略時は
            sourceと同じディレクトリに「utf-8」を付けた名前で保存。
        encoding: 入力ファイルのエンコーディング（既定: shift_jis）。
        output_encoding: 出力ファイルのエンコーディング（既定: utf-8）。

    Returns:
        書き出したファイルのパス。

    Raises:
        FileNotFoundError: 入力ファイルが存在しないとき。
        UnicodeDecodeError: 入力ファイルが指定エンコーディングで
            解釈できないとき。
        UnicodeEncodeError: 出力エンコーディングで文字化けするとき。
        csv.Error: CSVの構文エラーなど。
    """
    source_path = Path(source)

    if not source_path.is_file():
        raise FileNotFoundError(
            f"入力ファイルが見つかりません: {source_path}"
        )

    destination_path = Path(destination) if destination else (
        source_path.with_name(f"{source_path.stem}-utf-8.csv")
    )

    with source_path.open(encoding=encoding, newline="") as infile, \
            destination_path.open("w", encoding=output_encoding, newline="") as outfile:
        reader = csv.reader(infile)
        writer = csv.writer(outfile)
        for row in reader:
            writer.writerow(row)

    return destination_path
```

**使い方**

```python
from pathlib import Path
from convert_sjis_csv import convert_sjis_csv_to_utf8

src = Path("data/shift_jis.csv")
dst = convert_sjis_csv_to_utf8(src)          # data/shift_jis-utf-8.csv
dst2 = convert_sjis_csv_to_utf8(src, "data/out.csv")
```

**エラー処理の要点**

- `FileNotFoundError` — 存在しないファイルは、部分書き出しを避けるため先にチェック
- `UnicodeDecodeError` — 文字化け（デコード失敗）は `open(encoding=...)` で発生。`errors="replace"` 等にして黙らせるのは避けた
- `UnicodeEncodeError` — 出力側で文字化けした場合は半開きファイルが残らないよう、`with` 文内で閉じてから例外を送出
- `csv.Error` — 構文エラーもそのまま伝播

**補足**

- `newline=""` を指定しているのは、WindowsでCSVの改行が二重になるのを防ぐため
- 出力名は `source` の名前に `-utf-8` を付けたもの。必要なら `destination` で明示指定できます
- 出力先が既に存在する場合も上書きします。上書きを避けたい場合は `destination` を `Path(...).with_suffix(".csv")` などで指定してください
