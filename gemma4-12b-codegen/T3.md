## 応答全文 (そのまま)

ご要望の機能をすべて盛り込んだ、単一ファイルのHTMLコードを作成しました。

このコードには以下の機能が含まれています：
1.  **Canvas描画**: HTML5 Canvasを使用したスムーズなアニメーション。
2.  **天体システム**: 太陽を中心に地球が回り、地球の周りを月が回る構造。
3.  **正確な周期比**: 地球の公転1周に対し、月が約12回公転する計算式を実装。
4.  **軌道線**: 各天体の軌道を視覚的に分かりやすく描画。
5.  **操作系**: 一時停止/再開ボタンおよび3段階の速度調整（遅い・普通・速い）。
6.  **ラベル表示**: 各天体に名前を表示。

    <script>
        const canvas = document.getElementById('orbitCanvas');
        const ctx = canvas.getContext('2d');
        const pauseBtn = document.getElementById('pauseBtn');
        const speedSelect = document.getElementById('speedSelect');

        // 設定
        let width, height;
        let isPaused = false;
        let speedMultiplier = 1.0;

        // 天体のデータ
        const sun = { x: 0, y: 0, radius: 35, color: '#ffcc00', name: '太陽' };
        const earth = {
            orbitRadius: 200,
            angle: 0,
            size: 10,
            color: '#4a90e2',
            name: '地球'
        };
        const moon = {
            orbitRadius: 45, // 地球からの距離
            angle: 0,
            size: 4,
            color: '#d3d3d3',
            name: '月'
        };

        // キャンバスサイズ調整
        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
        }

        window.addEventListener('resize', resize);
        resize();

        // 操作系
        pauseBtn.addEventListener('click', () => {
            isPaused = !isPaused;
            pauseBtn.textContent = isPaused ? '再開' : '一時停止';
        });

        speedSelect.addEventListener('change', (e) => {
            speedMultiplier = parseFloat(e.target.value);
        });

        function draw() {
            ctx.clearRect(0, 0, width, height);
            
            const centerX = width / 2;
            const centerY = height / 2;

            // --- 1. 軌道線の描画 ---
            // 地球の軌道
            ctx.beginPath();
            ctx.arc(centerX, centerY, earth.orbitRadius, 0, Math.PI * 2);
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.15)';
            ctx.lineWidth = 1;
            ctx.stroke();

            // 月の軌道（地球の周り）
            ctx.beginPath();
            // 地球の現在の位置を計算して、その周りに円を描くための準備
            // ※動的な描画のため、ここでは「形」として描く
            ctx.arc(centerX, centerY, earth.orbitRadius, 0, Math.PI * 2); // これは地球軌道（再確認）
            
            // 正しい月の軌道の描き方：地球の周りの円
            // 動的な位置を計算する前に、幾何学的な形として描く。
            // 地球の軌道は固定なので、中心(centerX, centerY)からの距離 earth.orbitRadius を使う。
            // 月の軌道は「地球を中心とした半径 moon.orbitRadius」の円。
            // 実際には地球が動くので、このコードでは「地球の周りの軌道線」を常に描画する。
            ctx.beginPath();
            ctx.arc(centerX, centerY, earth.orbitRadius, 0, Math.PI * 2); // 地球軌道（再描画）
            ctx.stroke();

            // 月の軌道の描画（地球の周りの円を動的に追従させるのではなく、構造的な軌道を表現）
            // ここでは「地球の周囲にある月が通るパス」を描く。
            // 地球が移動しても、その位置を中心とした小さな円を描く。
            // しかし、このシミュレーションでは地球の軌道は固定なので、
            // 実際には「地球の位置を基準にした円」が必要。
            
            // --- 2. アニメーション計算 ---
            if (!isPaused) {
                // 地球の回転（1周に相当する角度を更新）
                earth.angle += 0.005 * speedMultiplier;
                
                // 月の回転（地球の1周に対し約12回。つまり、地球の速度の12倍速く動かす）
                moon.angle += (0.005 * 12) * speedMultiplier;
            }

            // 地球の座標計算
            const earthX = centerX + Math.cos(earth.angle) * earth.orbitRadius;
            const earthY = centerY + Math.sin(earth.angle) * earth.orbitRadius;

            // 月の座標計算（地球の位置を基準にする）
            const moonX = earthX + Math.cos(moon.angle) * moon.orbitRadius;
            const moonY = earthY + Math.sin(moon.angle) * moon.orbitRadius;

            // --- 3. 天体の描画 ---

            // 太陽の軌道（中心）
            ctx.beginPath();
            ctx.arc(centerX, centerY, sun.radius, 0, Math.PI * 2);
            ctx.fillStyle = sun.color;
            ctx.shadowBlur = 20;
            ctx.shadowColor = sun.color;
            ctx.fill();
            ctx.shadowBlur = 0; // 他の描画に影響させない

            // 地球の軌道（動的な位置に基づく月用）
            // 月が地球の周りを回るための「小さな円」を、地球の位置に合わせて描く。
            ctx.beginPath();
            ctx.arc(earthX, earthY, moon.orbitRadius, 0, Math.PI * 2);
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
            ctx.stroke();

            // 地球の描画
            ctx.beginPath();
            ctx.arc(earthX, earthY, earth.size, 0, Math.PI * 2);
            ctx.fillStyle = earth.color;
            ctx.fill();

            // 月の描画
            ctx.beginPath();
            ctx.arc(moonX, moonY, moon.size, 0, Math.PI * 2);
            ctx.fillStyle = moon.color;
            ctx.fill();

            // --- 4. ラベルの描画 ---
            ctx.fillStyle = "white";
            ctx.font = "14px Arial";
            ctx.textAlign = "center";

            ctx.fillText(sun.name, centerX, centerY + sun.radius + 20);
            ctx.fillText(earth.name, earthX, earthY + earth.size + 15);
            ctx.fillText(moon.name, moonX, moonY + moon.size + 10);

            requestAnimationFrame(draw);
        }

        draw();
    </script>
