<script>
    // --- State Management ---
    let todos = JSON.parse(localStorage.getItem('todos')) || [];
    let currentFilter = 'all';

    const todoInput = document.getElementById('todoInput');
    const addBtn = document.getElementById('addBtn');
    const todoList = document.getElementById('todoList');
    const incompleteCountSpan = document.getElementById('incompleteCount');
    const filterBtns = document.querySelectorAll('.filter-btn');

    // --- Core Functions ---

    // Save to LocalStorage
    function save() {
        localStorage.setItem('todos', JSON.stringify(todos));
        render();
    }

    // Render the UI
    function render() {
        // Update Count
        const incompleteCount = todos.filter(t => !t.completed).length;
        incompleteCountSpan.textContent = incompleteCount;

        // Filter Logic
        const filteredTodos = todos.filter(todo => {
            if (currentFilter === 'incomplete') return !todo.completed;
            if (currentFilter === 'completed') return todo.completed;
            return true;
        });

        // Render List
        todoList.innerHTML = '';
        filteredTodos.forEach((todo, index) => {
            const li = document.createElement('li');
            
            const checkbox = document.createElement('input');
            checkbox.type = 'checkbox';
            checkbox.checked = todo.completed;
            checkbox.onchange = () => {
                todo.completed = !todo.completed;
                save();
            };

            const span = document.createElement('span');
            span.className = `todo-text ${todo.completed ? 'completed' : ''}`;
            span.textContent = todo.text;
            // Double click to edit
            span.ondblclick = () => {
                const newText = prompt("タスクを編集してください:", todo.text);
                if (newText !== null && newText.trim() !== "") {
                    todo.text = newText.trim();
                    save();
                }
            };

            const delBtn = document.createElement('button');
            delBtn.textContent = '削除';
            delBtn.className = 'delete-btn';
            delBtn.onclick = () => {
                // Find original index in the full list (not just filtered)
                const actualIndex = todos.indexOf(todo);
                todos.splice(actualIndex, 1);
                save();
            };

            li.appendChild(checkbox);
            li.appendChild(span);
            li.appendChild(delBtn);
            todoList.appendChild(li);
        });
    }

    // Add Todo
    function addTodo() {
        const text = todoInput.value.trim();
        if (text) {
            todos.push({
                id: Date.now(),
                text: text,
                completed: false
            });
            todoInput.value = '';
            save();
        }
    }

    // Event Listeners
    addBtn.addEventListener('click', addTodo);

    todoInput.addEventListener('keypress', (e) => {
        if (e.key === 'Enter') addTodo();
    });

    filterBtns.forEach(btn => {
        btn.addEventListener('click', () => {
            // Update active button UI
            filterBtns.forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            
            // Change filter state
            currentFilter = btn.dataset.filter;
            render();
        });
    });

    // Initial Render
    render();
</script>

</body>

### 実装した機能の解説：
1.  **タスクの追加**: 入力欄に文字を入れて「追加」ボタンを押すか、Enterキーを押すことでリストに追加されます。
2.  **完了/未完了の切り替え**: 各項目の左側にあるチェックボックスを操作することで状態が変わり、見た目も打ち消し線で変化します。
3.  **削除機能**: 右側の「削除」ボタンを押すと、そのタスクが配列から取り除かれます。
4.  **ダブルクリックでの編集**: タスクのテキスト部分をダブルクリックするとブラウザの入力ダイアログ（prompt）が表示され、内容を書き換えることができます。
5.  **フィルタ機能**: 上部の「すべて」「未完了」「完了」ボタンで表示する項目を瞬時に切り替えます。
6.  **未完了カウント**: 常に現在の全データのうち、未完了のものがいくつあるかをリアルタイムに計算して表示します。
7.  **永続化（localStorage）**: タスクを追加・更新・削除するたびにブラウザのストレージに保存されるため、ページをリロードしても内容が保持されます。
