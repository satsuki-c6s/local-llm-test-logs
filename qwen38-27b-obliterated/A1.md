import csv
import os


def convert_encoding(input_path: str, output_path: str) -> None:
    """
    Shift-JISエンコードのCSVファイルを読み込み、UTF-8エンコードで保存する。

    Args:
        input_path: 入力ファイルのパス
        output_path: 出力ファイルのパス

    Raises:
        FileNotFoundError: 入力ファイルが存在しない場合
        UnicodeDecodeError: デコードに失敗した場合
    """
    if not os.path.exists(input_path):
        raise FileNotFoundError(f"ファイルが存在しません: {input_path}")

    try:
        with open(input_path, 'r', encoding='shift_jis') as f_in:
            reader = csv.reader(f_in)
            rows = list(reader)
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding, e.object, e.start, e.end,
            f"Shift-JISデコードに失敗しました: {input_path}"
        ) from e

    with open(output_path, 'w', encoding='utf-8', newline='') as f_out:
        writer = csv.writer(f_out)
        writer.writerows(rows)
