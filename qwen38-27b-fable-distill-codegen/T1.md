(function(){
  const TASKS_KEY = 'todo.tasks.v1';
  const FILTER_KEY = 'todo.filter.v1';

  const list = $('#list'), empty = $('#empty'),
        emptyKanji = $('#emptyKanji'), emptyMsg = $('#emptyMsg'),
        badge = $('#badge'), remaining = $('#remaining'), clearBtn = $('#clearDone'),

  let tasks = load();
  let filter = localStorage.getItem(FILTER_KEY) || 'all';

  function load(){
    try{
      const raw = JSON.parse(localStorage.getItem(TASKS_KEY));
      return Array.isArray(raw) ? raw.filter(t => t && typeof t.text === 'string') : [];
    }catch(e){ return []; }
  }
  function save(){ localStorage.setItem(TASKS_KEY, JSON.stringify(tasks)); }

  function visible(){
    return tasks.filter(t => filter === 'all' ? true : filter === 'active' ? !t.done : t.done);
  }

  function render(){
    list.innerHTML = '';
    visible().forEach((t, i) => {
      const li = makeItem(t);

  function makeItem(t){
    const li = document.createElement('li');
    li.className = 'task' + (t.done ? ' done' : '');
    li.dataset.id = t.id;

    const cb = document.createElement('input');
    cb.type = 'checkbox'; cb.className = 'check'; cb.checked = t.done;
    cb.setAttribute('aria-label', 'タスクを完了にする');
    cb.addEventListener('change', () => {
      t.done = cb.checked; save();
      li.classList.toggle('done', t.done);
      if (filter !== 'all'){
        collapse(li, () => { li.remove(); updateMeta(); });
      } else {
        updateMeta();
      }
    });

    const span = document.createElement('span');
    span.className = 'text'; span.textContent = t.text;
    span.title = 'ダブルクリックで編集';
    span.addEventListener('dblclick', () => startEdit(li, t, span));

    const del = document.createElement('button');
    del.className = 'del'; del.type = 'button'; del.textContent = '✕';
    del.setAttribute('aria-label', 'タスクを削除');
    del.addEventListener('click', () => {
      tasks = tasks.filter(x => x.id !== t.id); save();
      collapse(li, () => { li.remove(); updateMeta(); });
    });

    li.append(cb, span, del);
    return li;
  }

  function startEdit(li, t, span){
    if (li.querySelector('.edit-input')) return;
    const input = document.createElement('input');
    input.className = 'edit-input'; input.value = t.text;
    input.setAttribute('aria-label', 'タスクを編集');
    span.replaceWith(input);
    input.focus(); input.select();

    let finished = false;
    const restore = commit => {
      if (finished) return; finished = true;
      if (commit){
        const v = input.value.trim();
        if (v) t.text = v;
        save();
      }
      const s = document.createElement('span');
      s.className = 'text'; s.textContent = t.text; s.title = 'ダブルクリックで編集';
      s.addEventListener('dblclick', () => startEdit(li, t, s));
      input.replaceWith(s);
    };
    input.addEventListener('keydown', e => {
      if (e.key === 'Enter') restore(true);
      else if (e.key === 'Escape') restore(false);
    });
    input.addEventListener('blur', () => restore(true));
  }

  function collapse(li, cb){
    li.style.height = li.offsetHeight + 'px';
    li.style.overflow = 'hidden';
    li.style.pointerEvents = 'none';
    requestAnimationFrame(() => {
      li.style.transition = 'height .24s ease, opacity .18s ease, padding .24s ease, border-width .24s ease';

  });

    doneLis.forEach((li, i) => setTimeout(() => collapse(li, () => li.remove()), i * 70));

  });

  setFilter(filter, false);
  requestAnimationFrame(moveIndicator);

**要件への対応**

1. **追加** — 入力欄 + 「追加」ボタン（Enterキーでも追加、空入力時はシェイクで警告）
2. **完了切替** — 各タスクのチェックボックス（チェックマークの描画アニメーション付き）
3. **削除** — 各行の ✕ ボタン（収縮アニメーション付き）
4. **編集** — テキストをダブルクリックで編集モード、Enterで確定・Escでキャンセル・フォーカス外で確定
5. **フィルタ** — 「すべて / 未完了 / 完了」のセグメントコントロール（スライドインジケーター付き）
6. **件数表示** — ヘッダーのバッジとフッターの両方に未完了件数を表示
7. **永続化** — タスクとフィルタ状態を `localStorage` に保存し、リロード後も復元
