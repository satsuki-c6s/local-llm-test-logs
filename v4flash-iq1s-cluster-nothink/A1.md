```python
import csv
import os
from pathlib import Path

def convert_csv_shiftjis_to_utf8(input_path, output_path=None):
    """
    Shift-JISのCSVファイルを読み込み、UTF-8で保存し直す関数
    
    Parameters:
    -----------
    input_path : str or Path
        入力ファイルのパス（Shift-JISエンコーディング）
    output_path : str or Path, optional
        出力ファイルのパス。指定しない場合は入力ファイルと同じ場所に
        '_utf8' を付加した名前で保存する
    
    Returns:
    --------
    str : 出力ファイルのパス
    
    Raises:
    -------
    FileNotFoundError : 入力ファイルが存在しない場合
    UnicodeDecodeError : デコードに失敗した場合（文字化け）
    """
    
    # 入力ファイルの存在確認
    input_path = Path(input_path)
    if not input_path.is_file():
        raise FileNotFoundError(f"ファイルが存在しません: {input_path}")
    
    # 出力パスの設定
    if output_path is None:
        output_path = input_path.with_name(input_path.stem + '_utf8' + input_path.suffix)
    else:
        output_path = Path(output_path)
    
    try:
        # Shift-JISで読み込む
        with open(input_path, 'r', encoding='shift_jis', newline='') as f:
            reader = csv.reader(f)
            rows = list(reader)
        
        # UTF-8で書き出す
        with open(output_path, 'w', encoding='utf-8', newline='') as f:
            writer = csv.writer(f)
            writer.writerows(rows)
        
        print(f"変換完了: {input_path} -> {output_path}")
        return str(output_path)
    
    except UnicodeDecodeError as e:
        # デコード失敗時の詳細なエラーメッセージ
        raise UnicodeDecodeError(
            e.encoding, e.object, e.start, e.end, 
            f"Shift-JISデコードに失敗しました。ファイルに文字化けが含まれています。"
            f"エラー位置: {e.start}〜{e.end} (ファイル: {input_path})"
        ) from e

# 使用例
if __name__ == "__main__":
    # 正常な変換
    try:
        result = convert_csv_shiftjis_to_utf8("data.csv")
        print(f"変換成功: {result}")
    except FileNotFoundError as e:
        print(f"エラー: {e}")
    except UnicodeDecodeError as e:
        print(f"エラー: {e}")
    
    # 存在しないファイルの場合
    try:
        convert_csv_shiftjis_to_utf8("nonexistent.csv")
    except FileNotFoundError as e:
        print(f"エラー: {e}")
    
    # 文字化けが発生するファイルの場合（例として）
    # 実際にはShift-JISでないファイルを指定するとエラーになる
    try:
        convert_csv_shiftjis_to_utf8("utf8_file.csv")
    except UnicodeDecodeError as e:
        print(f"エラー: {e}")
```

**主なポイント:**

1. **ファイル存在確認**: `Path.is_file()` で事前に存在をチェック
2. **デコードエラー処理**: `UnicodeDecodeError` を捕捉し、エラー位置情報を付加
3. **CSVの改行対応**: `newline=''` を指定してCSVの改行を正しく処理
4. **出力パスの自動生成**: 入力ファイル名に `_utf8` を付加
5. **エラー情報の詳細化**: エラー位置をメッセージに含める

**注意点:**
- 実際の文字化けは、ファイルがShift-JISではない場合や、特殊な文字が含まれる場合に発生します
- エンコーディングを `'shift_jis'` から `'cp932'` に変更すると、より多くの文字を扱えます（Windows環境向け）
- 必要に応じて `csv.reader` の `delimiter` パラメータで区切り文字を指定できます
