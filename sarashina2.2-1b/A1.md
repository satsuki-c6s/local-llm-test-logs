```python
import csv
import sys

def convert_and_save_csv(input_file, output_file):
    # 入力ファイルの読み込みとUTF-8への変換
    try:
        with open(input_file, mode='r', encoding='shift_jis') as infile:
            reader = csv.reader(infile)
            writer = csv.writer(open(output_file, mode='w', encoding='utf-8'))
            
            # CSVの各行をUTF-8に変換して書き込む
            for row in reader:
                writer.writerow(row)
        print(f"CSV conversion and saving successful for {output_file}")
    except Exception as e:
        print(f"Error converting or saving the file: {e}")

# 使用例
input_file = "shiftjis_file.csv"
output_file = "utf8_converted_file.csv"
convert_and_save_csv(input_file, output_file)
```
