## プロンプト

```
ブラウザで動くToDoアプリを、外部ライブラリやCDNを使わず、単一のHTMLファイル1つだけで作ってください。要件: (1) テキスト入力と追加ボタンでタスクを追加できる (2) 各タスクはチェックで完了/未完了を切り替えられる (3) 各タスクは削除できる (4) 各タスクはダブルクリックで内容を編集できる (5) 「すべて/未完了/完了」の3つのフィルタ切り替えがある (6) 未完了タスクの件数が表示される (7) localStorageに保存され、リロードしても消えない。コードはHTMLファイル全文を1つのコードブロックで出力してください。
```

## 応答全文 (そのまま)

```html
<!DOCTYPE html>
<html>
<head>
  <title>Simple ToDo App</title>
  <style>
    body {
      font-family: sans-serif;
    }

    #todo-list {
      list-style: none;
      padding: 0;
    }

    #todo-list li {
      display: flex;
      align-items: center;
      margin-bottom: 5px;
      border: 1px solid #ccc;
      padding: 5px;
      background-color: #f9f9f9;
    }

    #todo-list li input[type="checkbox"] {
      margin-right: 10px;
    }

    #todo-list li.completed {
      text-decoration: line-through;
      background-color: #e6ffe6;
    }

    #todo-input {
      padding: 5px;
      border: 1px solid #ccc;
    }

    #add-button {
      padding: 5px 10px;
      background-color: #4CAF50;
      color: white;
      border: none;
      cursor: pointer;
    }

    #filters button {
      margin-right: 5px;
      padding: 5px 10px;
      cursor: pointer;
    }

    #incomplete-count {
      margin-top: 10px;
      font-weight: bold;
    }
  </style>
</head>
<body>

  <h1>Simple ToDo App</h1>

  <div>
    <input type="text" id="todo-input" placeholder="Add a new task">
    <button id="add-button">Add</button>
  </div>

  <div id="filters">
    <button data-filter="all">All</button>
    <button data-filter="incomplete">Incomplete</button>
    <button data-filter="completed">Completed</button>
  </div>

  <ul id="todo-list"></ul>

  <div id="incomplete-count">Uncompleted tasks: 0</div>


  <script>
    let todos = [];
    const todoInput = document.getElementById('todo-input');
    const addButton = document.getElementById('add-button');
    const todoList = document.getElementById('todo-list');
    const incompleteCountElement = document.getElementById('incomplete-count');
    const filters = document.getElementById('filters');

    // LocalStorageからToDoリストを読み込む
    function loadTodos() {
      const storedTodos = localStorage.getItem('todos');
      if (storedTodos) {
        todos = JSON.parse(storedTodos);
      }
      renderTodos();
    }

    // ToDoリストをLocalStorageに保存する
    function saveTodos() {
      localStorage.setItem('todos', JSON.stringify(todos));
    }


    function renderTodos() {
      todoList.innerHTML = '';
      const selectedFilter = Array.from(filters.children).find(button => button.classList.contains('active'))?.dataset.filter || 'all';

      let filteredTodos = todos;
      if (selectedFilter === 'incomplete') {
        filteredTodos = todos.filter(todo => !todo.completed);
      } else if (selectedFilter === 'completed') {
        filteredTodos = todos.filter(todo => todo.completed);
      }

      filteredTodos.forEach((todo, index) => {
        const li = document.createElement('li');
        li.classList.add(todo.completed ? 'completed' : '');

        const checkbox = document.createElement('input');
        checkbox.type = 'checkbox';
        checkbox.checked = todo.completed;
        checkbox.addEventListener('change', () => {
          todo.completed = !todo.completed;
          li.classList.toggle('completed');
          saveTodos();
          updateIncompleteCount();
          renderTodos(); //再描画してチェックボックスの状態を更新
        });

        const span = document.createElement('span');
        span.textContent = todo.text;
        span.addEventListener('dblclick', () => {
          const newText = prompt('Edit task:', todo.text);
          if (newText !== null) {
            todo.text = newText;
            saveTodos();
            renderTodos();
          }
        });

        const deleteButton = document.createElement('button');
        deleteButton.textContent = 'Delete';
        deleteButton.addEventListener('click', () => {
          todos = todos.filter((_, i) => i !== index);
          saveTodos();
          updateIncompleteCount();
          renderTodos();
        });

        li.appendChild(checkbox);
        li.appendChild(span);
        li.appendChild(deleteButton);
        todoList.appendChild(li);
      });

      updateIncompleteCount();
    }


    function updateIncompleteCount() {
      const incompleteTasks = todos.filter(todo => !todo.completed);
      incompleteCountElement.textContent = `Uncompleted tasks: ${incompleteTasks.length}`;
    }



    addButton.addEventListener('click', () => {
      const text = todoInput.value.trim();
      if (text !== '') {
        todos.push({ text, completed: false });
        saveTodos();
        todoInput.value = '';
        renderTodos();
      }
    });

    filters.addEventListener('click', (event) => {
      if (event.target.tagName === 'BUTTON') {
        // すべてのボタンから'active'クラスを削除
        Array.from(filters.children).forEach(button => button.classList.remove('active'));
        // クリックされたボタンに'active'クラスを追加
        event.target.classList.add('active');
        renderTodos();
      }
    });

    loadTodos(); // 初期ロード時にToDoリストを読み込む
  </script>

</body>
</html>
```
