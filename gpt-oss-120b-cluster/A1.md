## 🛠️ 実装例

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Shift‑JIS の CSV を UTF‑8 に変換して保存するユーティリティ。

使い方
------
>>> from shiftjis2utf8 import convert_shiftjis_to_utf8
>>> convert_shiftjis_to_utf8("data_sjis.csv", "data_utf8.csv")
"""

import os
import csv
from typing import Iterable, List, Sequence

# ------------------------------------------------------------
# 1️⃣ カスタム例外（エラーの種類が分かりやすくなる）
# ------------------------------------------------------------
class EncodingError(RuntimeError):
    """Shift‑JIS デコードに失敗したときに投げられる例外。"""
    pass


# ------------------------------------------------------------
# 2️⃣ CSV を Shift‑JIS で読み込み、UTF‑8 で書き出す本体
# ------------------------------------------------------------
def _read_shiftjis_csv(path: str) -> List[List[str]]:
    """
    Shift‑JIS エンコーディングで CSV を読み込み、行ごとのリストを返す。

    Parameters
    ----------
    path : str
        読み込む CSV ファイルへのフルパス。

    Returns
    -------
    List[List[str]]
        1 行ごとに分割された文字列のリスト。
    """
    # 1. ファイルが本当に存在するかチェック
    if not os.path.isfile(path):
        raise FileNotFoundError(f"ファイルが見つかりません: {path}")

    # 2. デコードエラーを捕捉したいので、strict モードで開く
    try:
        with open(path, "r", encoding="shift_jis", newline="") as f:
            # csv.reader はテキストストリームを受け取るのでそのまま使える
            reader = csv.reader(f)
            rows = [list(row) for row in reader]   # 1 行ずつ取得
        return rows
    except UnicodeDecodeError as e:
        # デコードに失敗したら独自例外でラップして再送出
        raise EncodingError(
            f"Shift‑JIS としてデコードできませんでした (ファイル: {path})\n"
            f"エラー詳細: {e}"
        ) from e


def _write_utf8_csv(rows: Sequence[Sequence[str]], dst_path: str) -> None:
    """
    受け取った行データを UTF‑8 エンコーディングで CSV として書き出す。

    Parameters
    ----------
    rows : Sequence[Sequence[str]]
        2 次元リスト（行 → 列）形式のデータ。
    dst_path : str
        書き出す先ファイル名（フルパス）。
    """
    # 書き込みは必ず UTF‑8 (BOM なし) で行う
    with open(dst_path, "w", encoding="utf-8", newline="") as f:
        writer = csv.writer(f)
        writer.writerows(rows)


# ------------------------------------------------------------
# 3️⃣ 公開 API
# ------------------------------------------------------------
def convert_shiftjis_to_utf8(src_path: str, dst_path: str) -> None:
    """
    Shift‑JIS の CSV を UTF‑8 に変換して保存する。

    Parameters
    ----------
    src_path : str
        変換元（Shift‑JIS）CSV のパス。
    dst_path : str
        変換後に保存したい UTF‑8 CSV のパス。

    例外
    -----
    FileNotFoundError
        src_path が存在しない場合に送出。
    EncodingError
        Shift‑JIS デコードに失敗した場合に送出。
    """
    # 1) Shift‑JIS で読み込む
    rows = _read_shiftjis_csv(src_path)

    # 2) UTF‑8 で書き出す
    _write_utf8_csv(rows, dst_path)


# ------------------------------------------------------------
# 4️⃣ 直接実行したときのサンプル CLI
# ------------------------------------------------------------
if __name__ == "__main__":
    import argparse
    import sys

    parser = argparse.ArgumentParser(
        description="Shift‑JIS の CSV を UTF‑8 に変換して保存する"
    )
    parser.add_argument("src", help="変換元 CSV (Shift‑JIS)")
    parser.add_argument("dst", help="変換後に保存する CSV (UTF‑8)")
    args = parser.parse_args()

    try:
        convert_shiftjis_to_utf8(args.src, args.dst)
        print(f"変換完了: {args.src} → {args.dst}")
    except (FileNotFoundError, EncodingError) as e:
        sys.stderr.write(f"エラー: {e}\n")
        sys.exit(1)
```

## 🔎 主要ポイントの解説

| 項目 | 内容 |
|------|------|
| **エンコーディング指定** | `open(..., encoding="shift_jis")` と `encoding="utf-8"` を明示的に指定。 |
| **エラー検出** | `open(..., encoding="shift_jis", ...)` で **strict**（デフォルト）モードにしているので、デコードできないバイト列が出たら `UnicodeDecodeError` が発生します。 |
| **独自例外** | `EncodingError` を作って、呼び出し側が「ファイルはあるが文字化けした」ケースと「ファイルが無い」ケースを簡単に判別できるようにしています。 |
| **CSV の扱い** | `csv.reader` / `csv.writer` は **テキストストリーム** を受け取るので、エンコーディングは `open` のみで OK。 |
| **改行コード** | `newline=""` を指定することで、プラットフォーム依存の改行変換を防ぎ、CSV の行区切りが正しく保たれます。 |
| **BOM の有無** | `utf-8` で書き込むときは **BOM なし** が一般的です。必要なら `encoding="utf-8-sig"` に変更してください。 |
| **CLI** | スクリプト単体でも `python convert_shiftjis_to_utf8.py src.csv dst.csv` と実行でき、エラーメッセージが分かりやすく出ます。 |

## 📦 使い方例

```python
from shiftjis2utf8 import convert_shiftjis_to_utf8, EncodingError

src = "sample_shiftjis.csv"
dst = "sample_utf8.csv"

try:
    convert_shiftjis_to_utf8(src, dst)
    print("変換が成功しました！")
except FileNotFoundError:
    print(f"ファイルが見つかりません: {src}")
except EncodingError as e:
    print(f"文字化けが疑われます: {e}")
```
