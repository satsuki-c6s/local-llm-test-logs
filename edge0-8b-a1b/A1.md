```python
import csv
import os
from pathlib import Path

def convert_sjis_to_utf8(input_path: str, output_path: str) -> bool:
    """
    Shift-JISのCSVファイルをUTF-8に変換して保存する関数。
    
    Parameters:
        input_path: ソースファイルのパス
        output_path: 出力ファイルのパス
    
    Returns:
        成功した場合はTrue、失��した場合はFalse
    
    Raises:
        FileNotFoundError: ソースファイルが存在しない場合
        ValueError: 変換中に発生した文字化けエラー
        Exception: その他の予想外のエラー
    """
    # 1. ソースファイルの存在チェック
    if not os.path.exists(input_path):
        raise FileNotFoundError(f"ソースファイルが見つかりません: {input_path}")
    
    # 2. 出力ディレクトリが存在しない場合作成
    output_dir = os.path.dirname(output_path)
    if output_dir and not os.path.exists(output_dir):
        os.makedirs(output_dir, exist_ok=True)
    
    try:
        # 3. ソースファイルを「Shift-JIS」として読み込み
        with open(input_path, 'r', encoding='shift_jis') as src_file:
            reader = csv.reader(src_file)
            
            # 4. UTF-8形式で書き込み
            with open(output_path, 'w', encoding='utf-8') as dst_file:
                writer = csv.writer(dst_file)
                
                for row in reader:
                    # 書き込み前に、行内の特殊文字を正規化（例えば、\n → \r\n等）
                    # または各文字を明示的に再コード化（こちらの方が安全）
                    normalized_row = [c.encode('utf-8').decode('utf-8') for c in row]
                    writer.writerow(normalized_row)
        
        return True
        
    except UnicodeDecodeError as e:
        raise ValueError(f"文字化けエラーが発生しました:\n{e}")
        
    except Exception as e:
        raise Exception(f"変換中に予期せぬエラーが発生しました:\n{type(e).__name__}: {e}")


# --- 使用例 ---
if __name__ == "__main__":
    source_file = "data_input.csv"   # Shift-JISの元CSVファイル
    target_file = "data_output.csv"  # UTF-8形式の出力CSVファイル
    
    try:
        success = convert_sjis_to_utf8(source_file, target_file)
        if success:
            print(f"変換完了: {target_file} を生成しました")
    except FileNotFoundError as fn_err:
        print(f"[エラー] ファイルがありません: {fn_err}")
    except ValueError as ve_err:
        print(f"[エラー] 文字化け: {ve_err}")
    except Exception as ex:
        print(f"[エラー] 予期せぬ事象: {ex}")
```

### 主要な機能まとめ

| 項目 | 説明 |
|------|------|
| **ファイル存在チェック** | `os.path.exists()` でソースファイルを確認 |
| **出力フォートルリング** | 親ディレクトリがあれば自動作成 |
| **Shift-JIS読み込み** | `encoding='shift_jis'` で正確にデコード |
| **UTF-8書き込み** | `encoding='utf-8'` で保存 |
| **エラー処理** | `UnicodeDecodeError` → 文字化けエラー、その他 → 例外 |
| **回復性** | 変換途中でエラーが発生しても、元ファイルを損傷しない |

> **注意**: もしShift-JISの特殊文字（シンタック・コントロール文字など）を含む場合、`encoding='shift_jis'` が正しくデコードできない可能性があります。その場合は、`chardkit` や `unicodedata` ライグロップを追加することをお��めします。
