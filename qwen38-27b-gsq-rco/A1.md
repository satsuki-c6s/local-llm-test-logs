```python
import csv
import os
import sys


def convert_shiftjis_to_utf8(input_path: str, output_path: str) -> None:
    """
    Shift-JISエンコードのCSVファイルを読み込み、UTF-8で保存し直す。

    Args:
        input_path: 入力ファイルのパス (Shift-JIS)
        output_path: 出力ファイルのパス (UTF-8)

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: Shift-JISデコードに失敗した場合
        csv.Error: CSV解析エラーの場合
    """
    # 1. ファイルの存在確認
    if not os.path.exists(input_path):
        raise FileNotFoundError(f"ファイルが存在しません: {input_path}")

    # 2. Shift-JISで読み込み
    try:
        with open(input_path, mode='r', encoding='shift_jis', newline='') as f_in:
            reader = csv.reader(f_in)
            rows = list(reader)
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding,
            e.object,
            e.start,
            e.end,
            f"Shift-JISデコードに失敗しました: {e.reason} (ファイル: {input_path})"
        )
    except csv.Error as e:
        raise csv.Error(f"CSV解析エラー: {e} (ファイル: {input_path})")

    # 3. UTF-8で書き込み
    try:
        with open(output_path, mode='w', encoding='utf-8', newline='') as f_out:
            writer = csv.writer(f_out)
            writer.writerows(rows)
    except OSError as e:
        raise OSError(f"出力ファイルの書き込みに失敗しました: {e} (ファイル: {output_path})")

    print(f"変換完了: {input_path} -> {output_path} ({len(rows)}行)")


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("使い方: python convert_csv.py <入力ファイル> <出力ファイル>")
        sys.exit(1)

    try:
        convert_shiftjis_to_utf8(sys.argv[1], sys.argv[2])
    except FileNotFoundError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
    except UnicodeDecodeError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
    except csv.Error as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
    except OSError as e:
        print(f"エラー: {e}", file=sys.stderr)
        sys.exit(1)
```

### 主なポイント

| 項目 | 説明 |
|------|------|
| **ファイル存在確認** | `os.path.exists()` で事前にチェックし、`FileNotFoundError` を明示的に送出 |
| **デコード失敗処理** | `UnicodeDecodeError` をキャッチし、元の理由とファイル名を含むメッセージで再送出 |
| **CSV解析エラー** | `csv.Error` をキャッチして処理 |
| **書き込みエラー** | `OSError` をキャッチして処理 |
| **`newline=''`** | `csv` モジュールの推奨設定。改行の二重処理を防止 |
| **`encoding='shift_jis'`** | Pythonの標準エンコーディング名（`cp932` と同義） |

### 使用例

### 補足：より堅牢にする場合

- **BOM付きUTF-8** で保存したい場合は `encoding='utf-8-sig'` を使用
- **エラーを無視してスキップ** したい場合は `errors='replace'` を `open()` に追加可能（ただしデータが失われるため注意）
- 大規模ファイルでは `list(reader)` による全量メモリ読み込みを避け、ストリーム処理に切り替える
