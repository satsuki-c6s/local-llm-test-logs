#### 1. SQLAlchemy (最も一般的)
Python で事実上の標準 ORM です。「SQLAlchemy」が「Sakura」と似ている可能性は低いですし、「sakura-orm」という名前のライブラリが存在するとは考えにくいため、もし名前が違う場合はこちらを推奨します。

**基本の使い方（コード例）**

```python
from sqlalchemy import create_engine, Column, Integer, String
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

# 1. エンジンとベースクラスを作成
engine = create_engine('sqlite:///example.db', echo=True) # デバッグ出力を有効化
Base = declarative_base()

# 2. モデル定義 (User テーブルの作成宣言)
class User(Base):
    __tablename__ = 'users'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(50))
    email = Column(String(120), unique=True)

# 3. DB にテーブルを作成 (初期化)
Base.metadata.create_all(engine)

# 4. セッション作成
SessionLocal = sessionmaker(bind=engine)
session = SessionLocal()

try:
    # --- データの追加 ---
    new_user = User(name="田中太郎", email="tanaka@example.com")
    session.add(new_user)
    session.commit()  # トランザクションコミット
    
    print(f"ID が割り当てられたユーザー：{new_user.id}")

    # --- データの取得 (クエリ) ---
    users = session.query(User).filter_by(name="田中太郎").all()
    
    if users:
        for user in users:
            print(f"{user.name} ({user.email})")

    # --- 更新と削除 ---
    target_user = session.get(1) # ID=1 のユーザーを取得 (存在すれば)
    if target_user:
        target_user.email = "updated@example.com"
        session.commit()
        
finally:
    session.close()
```

#### 2. Peewee (軽量でシンプル)
SQLAlchemy よりもコードが簡潔な ORM です。名前との関連性は薄いです。

**基本の使い方（コード例）**

```python
from peewee import Model, CharField, Integer, Database

# DB インスタンス作成 (メモリ上の SQLite を使用)
db = Database()

class User(Model):
    name = CharField(unique=True)
    email = CharField(null=False)

# テーブルの初期化
User.create_table(fields=[('id', 'auto'), ('name',), ('email',)])

try:
    # 追加
    user = User(name='山田花子', email='hanako@example.com')
    user.save()
    
    print(f"保存された ID: {user.id}")

    # クエリ (SELECT)
    users = User.select().where(User.name == '山田花子').execute()
    if not users:
        pass 
    else:
        for u in users:
            print(u.email)

finally:
    db.close()
```

#### 3. Django ORM (Web フレームワーク内蔵)
もし「Django」を使っている場合、ORM はフレームワークの一部として含まれています。名前が異なりますが、よく使われます。

**基本の使い方（コード例）**

```python
from django.db import models, connection # モデル定義とクエリ実行用

# 1. モデル定義 (models.py など)
class Product(models.Model):
    name = models.CharField(max_length=200)
    price = models.IntegerField()

    def __str__(self):
        return self.name

# Django の管理コマンドで migrate を走らせてテーブル作成後、以下を実行:

from django.db import connection, transaction # 簡易的なクエリ用 (または ORM API)
with connection.cursor() as cursor:
    # SQL ステートメントを直接実行することも可能だが、ORM は models.objects で行うのが一般的
    
# Django の ORM API を使う場合（Python コードから）:
from django.db import transaction

@transaction.atomic
def add_product():
    Product.objects.create(name="iPhone", price=120000) # 追加 (INSERT)
    
    p = Product.objects.get(id=1) # 取得 (SELECT)
    print(f"価格：{p.price}")
    
    p.price += 5000 # 更新準備
    p.save() # COMMIT を行う

add_product()
```
