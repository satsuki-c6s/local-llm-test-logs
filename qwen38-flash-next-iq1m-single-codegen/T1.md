function uid() {
  if (window.crypto && typeof window.crypto.randomUUID === 'function') return window.crypto.randomUUID();
  return `${Date.now()}-${Math.random().toString(16).slice(2)}`;
}

function escapeHtml(value) {
  return String(value).replace(/[&<>"']/g, (char) => {
    switch (char) {
      case '&': return '&amp;';
      case '<': return '&lt;';
      case '>': return '&gt;';
      case '"': return '&quot;';
      case "'": return '&#39;';
    }
  });
}

function addTask(text) {
  const trimmed = text.trim();
  if (!trimmed) return;
  tasks.unshift({ id: uid(), text: trimmed, completed: false });
  saveTasks();
  render();
}

function toggleTask(id) {
  const task = tasks.find((t) => t.id === id);
  if (!task) return;
  task.completed = !task.completed;
  saveTasks();
  render();
}

function deleteTask(id) {
  tasks = tasks.filter((t) => t.id !== id);
  if (editingId === id) {
    editingId = null;
    editingOriginal = '';
  }
  saveTasks();
  render();
}

function startEdit(id) {
  const task = tasks.find((t) => t.id === id);
  if (!task) return;
  editingId = id;
  editingOriginal = task.text;
  render();
  const editInput = list.querySelector('.edit-input');
  if (editInput) {
    editInput.focus();
    editInput.select();
  }
}

function replaceEditInputWithSpan(id, text) {
  const li = list.querySelector(`li[data-id="${id}"]`);
  const input = li ? li.querySelector('.edit-input') : null;
  if (!input) return;
  const span = document.createElement('span');
  span.className = 'task-text';
  span.textContent = text;
  input.replaceWith(span);
}

function saveEdit(id, value, rerender = true) {
  const task = tasks.find((t) => t.id === id);
  if (!task) {
    editingId = null;
    editingOriginal = '';
    if (rerender) render();
    return;
  }

  if (!trimmed) {

}
