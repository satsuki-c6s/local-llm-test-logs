## 代わりに：一般的なPython ORMの基本的な使い方

もしPythonでORMを使いたい場合は、以下のような人気ライブラリがあります。

### 1. SQLAlchemy（最も人気）

```python
# インストール
# pip install sqlalchemy

from sqlalchemy import create_engine, Column, Integer, String, ForeignKey
from sqlalchemy.orm import declarative_base, relationship, sessionmaker

# データベース接続
engine = create_engine("sqlite:///example.db")
Base = declarative_base()

# モデル定義
class User(Base):
    __tablename__ = 'users'

    id = Column(Integer, primary_key=True)
    name = Column(String, nullable=False)
    email = Column(String, unique=True)

    # リレーション
    posts = relationship("Post", back_populates="user")

class Post(Base):
    __tablename__ = 'posts'

    id = Column(Integer, primary_key=True)
    title = Column(String, nullable=False)
    content = Column(String)
    user_id = Column(Integer, ForeignKey('users.id'))

    user = relationship("User", back_populates="posts")

# テーブル作成
Base.metadata.create_all(engine)

# セッション作成
Session = sessionmaker(bind=engine)
session = Session()

# --- CRUD操作 ---

# Create（作成）
new_user = User(name="山田太郎", email="taro@example.com")
session.add(new_user)
session.commit()

# Read（読み取り）
users = session.query(User).all()
for user in users:
    print(f"ID: {user.id}, Name: {user.name}, Email: {user.email}")

# Update（更新）
user = session.query(User).filter_by(name="山田太郎").first()
if user:
    user.email = "taro.new@example.com"
    session.commit()

# Delete（削除）
if user:
    session.delete(user)
    session.commit()

session.close()
```

### 2. Peewee（軽量ORM）

```python
# インストール
# pip install peewee

from peewee import SqliteDatabase, Model, CharField, IntegerField, ForeignKeyField

db = SqliteDatabase('example.db')

class BaseModel(Model):
    class Meta:
        database = db

class User(BaseModel):
    name = CharField()
    email = CharField(unique=True)

class Post(BaseModel):
    title = CharField()
    content = CharField()
    user = ForeignKeyField(User, backref='posts')

# テーブル作成
db.connect()
db.create_tables([User, Post])

# Create
user = User.create(name="山田太郎", email="taro@example.com")

# Read
users = User.select()
for user in users:
    print(f"ID: {user.id}, Name: {user.name}")

# Update
user = User.get(User.name == "山田太郎")
user.email = "taro.new@example.com"
user.save()

# Delete
user.delete_instance()

db.close()
```

### 3. Tortoise ORM（非同期対応）

```python
# インストール
# pip install tortoise-orm

from tortoise import Tortoise, fields
from tortoise.models import Model

class User(Model):
    id = fields.IntField(pk=True)
    name = fields.CharField(max_length=255)
    email = fields.CharField(max_length=255, unique=True)

    class Meta:
        table = "users"

async def main():
    await Tortoise.init(db_url="sqlite://:memory:", modules={"models": ["__main__"]})
    await Tortoise.generate_schemas()

    # Create
    user = await User.create(name="山田太郎", email="taro@example.com")

    # Read
    users = await User.all()
    for user in users:
        print(f"ID: {user.id}, Name: {user.name}")

    # Update
    user = await User.get(name="山田太郎")
    user.email = "taro.new@example.com"
    await user.save()

    # Delete
    await user.delete()

    await Tortoise.close_connections()

import asyncio
asyncio.run(main())
```
