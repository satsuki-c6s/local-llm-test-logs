"""
Shift-JIS CSV ファイルを UTF-8 で再エンコードする関数。
存在しないファイル、文字化け（デコード失敗）のいずれかのエラーを
適切に処理し、エラーメッセージを返す。
"""

import csv
import io
import os
from typing import Optional, Tuple


def shiftjis_to_utf8(csv_path: str) -> Tuple[str, Optional[str]]:
    """
    Shift-JIS CSV ファイルを UTF-8 で再エンコードする。

    Args:
        csv_path: 入力ファイルのパス。

    Returns:
        (utf8_path, error_message):
            - success: 成功時のファイルパスと None
            - failure: 失敗時のファイルパスとエラーメッセージ
    """
    # 存在しないファイルのチェック
    if not os.path.exists(csv_path):
        return csv_path, f"ファイルが存在しません: {csv_path}"

    # 文字化け（デコード失敗）のチェック
    try:
        with open(csv_path, "r", encoding="shift_jis") as infile:
            content = infile.read()

        # CSV 形式のチェック
        lines = content.split("\n")
        if not lines:
            return csv_path, "ファイルは空です。"

        # 各行の文字列を UTF-8 で再エンコード
        utf8_lines = []
        for line in lines:
            try:
                utf8_line = line.encode("utf-8")
                utf8_lines.append(utf8_line)
            except UnicodeDecodeError:
                # 文字化けが発生した行を保持して、エラーを返す
                utf8_lines.append(line)

        if not utf8_lines:
            return csv_path, "ファイルは空です。"

        # 再エンコードした CSV ファイルを生成
        output_path = csv_path.replace(".csv", ".utf8.csv")
        with open(output_path, "w", encoding="utf-8", newline="") as outfile:
            writer = csv.writer(outfile)
            for line in utf8_lines:
                writer.writerow(line)

        return output_path, None

    except UnicodeDecodeError as e:
        # 文字化けが発生した行を保持して、エラーを返す
        return csv_path, f"文字化けが発生しました: {e}"

    except PermissionError:
        return csv_path, f"ファイルへの読み込み権限がありません: {csv_path}"
    except OSError as e:
        return csv_path, f"ファイル読み込みエラー: {e}"

## 主な機能とエラー処理

| エラーの種類 | 発生条件 | 処理方法 |
|---|---|---|
| **ファイル不存在** | 入力パスが存在しない | 存在確認 → 適切なエラーメッセージ返却 |
| **文字化け（UnicodeDecodeError）** | 行の文字列が Shift-JIS だが、UTF-8 で再エンコードできない | 文字化けした行を保持 → エラーを返す |
| **権限不足（PermissionError）** | ファイルに読み込み権限がない | 権限確認 → 適切なエラーメッセージ返却 |
| **OS エラー** | ファイルシステム上のエラー | 標準エラーメッセージ返却 |
