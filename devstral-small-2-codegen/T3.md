## 応答全文 (そのまま)

以下は、要求に応じた単一のHTMLファイルです。外部ライブラリやCDNを使用せず、Canvasで地球・太陽・月の軌道シミュレーターを実装しています。

```html
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>地球・太陽・月の軌道シミュレーター</title>
    <style>
        body {
            font-family: Arial, sans-serif;
            margin: 0;
            padding: 20px;
            background-color: #f0f0f0;
        }
        canvas {
            border: 1px solid #333;
            background-color: white;
            display: block;
            margin: 0 auto;
        }
        .controls {
            text-align: center;
            margin-bottom: 20px;
        }
        button {
            padding: 8px 16px;
            margin: 0 5px;
            font-size: 14px;
            cursor: pointer;
        }
    </style>
</head>
<body>
    <div class="controls">
        <button id="pauseBtn">一時停止</button>
        <button id="slowBtn">遅い</button>
        <button id="normalBtn">普通</button>
        <button id="fastBtn">速い</button>
    </div>
    <canvas id="orbitCanvas" width="800" height="600"></canvas>

    <script>
        // Canvas要素とコンテキストを取得
        const canvas = document.getElementById('orbitCanvas');
        const ctx = canvas.getContext('2d');

        // シミュレーションの状態
        let isPaused = false;
        let speed = 1; // 0.5 (遅い), 1 (普通), 2 (速い)

        // 天体のデータ
        const bodies = {
            sun: { radius: 30, color: '#FFD700', x: canvas.width / 2, y: canvas.height / 2 },
            earth: { radius: 15, color: '#1E90FF', orbitRadius: 200, angle: 0, speed: 0.01 * speed },
            moon: {
                radius: 7,
                color: '#A9A9A9',
                orbitRadius: 60,
                earthAngle: 0,
                earthSpeed: 0.04 * speed // 地球の公転に対して約12倍速い
            }
        };

        // ボタンイベントリスナー
        document.getElementById('pauseBtn').addEventListener('click', () => {
            isPaused = !isPaused;
            document.getElementById('pauseBtn').textContent = isPaused ? '再開' : '一時停止';
        });

        document.getElementById('slowBtn').addEventListener('click', () => { speed = 0.5; bodies.earth.speed = 0.01 * speed; bodies.moon.earthSpeed = 0.04 * speed; });
        document.getElementById('normalBtn').addEventListener('click', () => { speed = 1; bodies.earth.speed = 0.01 * speed; bodies.moon.earthSpeed = 0.04 * speed; });
        document.getElementById('fastBtn').addEventListener('click', () => { speed = 2; bodies.earth.speed = 0.01 * speed; bodies.moon.earthSpeed = 0.04 * speed; });

        // 天体を描画する関数
        function drawBody(body) {
            ctx.beginPath();
            ctx.arc(body.x, body.y, body.radius, 0, Math.PI * 2);
            ctx.fillStyle = body.color;
            ctx.fill();
            ctx.closePath();

            // ラベルを表示
            ctx.fillStyle = 'black';
            ctx.font = '12px Arial';
            ctx.textAlign = 'center';
            ctx.fillText(body.name, body.x, body.y + body.radius + 15);
        }

        // 軌道を描画する関数
        function drawOrbit(centerX, centerY, radius) {
            ctx.beginPath();
            ctx.arc(centerX, centerY, radius, 0, Math.PI * 2);
            ctx.strokeStyle = '#ccc';
            ctx.lineWidth = 1;
            ctx.stroke();
        }

        // アニメーションフレーム
        function animate() {
            if (!isPaused) {
                // 地球の角度を更新
                bodies.earth.angle += bodies.earth.speed;

                // 月の地球中心角度を更新（地球の公転に対して約12倍速い）
                bodies.moon.earthAngle += bodies.moon.earthSpeed;

                // 地球と月の位置を計算
                const earthX = bodies.sun.x + Math.cos(bodies.earth.angle) * bodies.earth.orbitRadius;
                const earthY = bodies.sun.y + Math.sin(bodies.earth.angle) * bodies.earth.orbitRadius;

                const moonX = earthX + Math.cos(bodies.moon.earthAngle) * bodies.moon.orbitRadius;
                const moonY = earthY + Math.sin(bodies.moon.earthAngle) * bodies.moon.orbitRadius;

                // 天体の位置を更新
                bodies.earth.x = earthX;
                bodies.earth.y = earthY;
                bodies.moon.x = moonX;
                bodies.moon.y = moonY;
            }

            // Canvasをクリア
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // 軌道を描画（太陽中心の地球軌道と地球中心の月軌道）
            drawOrbit(bodies.sun.x, bodies.sun.y, bodies.earth.orbitRadius);
            drawOrbit(earthX, earthY, bodies.moon.orbitRadius);

            // 天体を描画
            bodies.sun.name = '太陽';
            bodies.earth.name = '地球';
            bodies.moon.name = '月';

            drawBody(bodies.sun);
            drawBody(bodies.earth);
            drawBody(bodies.moon);

            requestAnimationFrame(animate);
        }

        // アニメーションを開始
        animate();
    </script>
</body>
</html>
```

このコードは以下の要件を満たしています：
1. Canvasで描画されています
2. 中央に太陽が配置され、その周りを地球が公転します
3. 月は地球の周りを公転しながら、地球と一緒に太陽の周りを回ります
4. 地球の公転1周の間に月が約12回公転する周期比になっています
5. 軌道線が表示されています
6. 一時停止/再開ボタンと速度調整（遅い/普通/速い）機能があります
7. 天体名のラベルが表示されています

コードは単一のHTMLファイルで完結しており、外部ライブラリやCDNを使用していません。
