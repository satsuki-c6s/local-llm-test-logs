#### SQLAlchemy の基本的な使い方（コード例）

この例では、`User` というモデルを定義し、データベースへの登録・取得を行います。

```python
from sqlalchemy import create_engine, Column, Integer, String
from sqlalchemy.orm import declarative_base, sessionmaker

# 1. データベースの接続設定 (今回は SQLite を使用)
DATABASE_URL = "sqlite:///sakura_users.db" # ファイル名は任意です
engine = create_engine(DATABASE_URL, echo=True) # echo=True で SQL ログを出力

# 2. ベースクラスの作成（モデル定義のための基底クラス）
Base = declarative_base()

# 3. モデルの定義 (User テーブルに対応する Python クラス)
class User(Base):
    __tablename__ = "users" # データベース上のテーブル名
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    name = Column(String(50), nullable=False)
    email = Column(String(100), unique=True, nullable=False)

# テーブルの作成 (存在しない場合のみ自動生成)
Base.metadata.create_all(engine)

# 4. セッションの準備
SessionLocal = sessionmaker(bind=engine)
session = SessionLocal()

try:
    # --- CREATE (登録) ---
    new_user1 = User(name="Sakura Tanaka", email="sakura@example.com")
    new_user2 = User(name="Yuki Sato", email="yuki@example.com")
    
    session.add(new_user1)
    session.add(new_user2)
    session.commit() # 変更を確定
    
    print("ユーザーが正常に登録されました。")

    # --- READ (検索) ---
    # name が "Sakura Tanaka" のユーザーを取得
    user = session.query(User).filter_by(name="Sakura Tanaka").first()
    
    if user:
        print(f"取得したユーザー: {user.name} ({user.email})")

    # --- UPDATE (更新) ---
    # 検索結果の email を変更
    user.email = "sakura.tanaka@newdomain.com"
    session.commit()
    print("メールアドレスが更新されました。")

    # --- DELETE (削除) ---
    # Yuki Sato を削除
    delete_target = session.query(User).filter_by(name="Yuki Sato").first()
    if delete_target:
        session.delete(delete_target)
        session.commit()
        print("ユーザーが削除されました。")

except Exception as e:
    session.rollback() # エラー発生時はロールバック（元に戻す）
    print(f"エラーが発生しました：{e}")

finally:
    session.close() # セッションを閉じる
```
