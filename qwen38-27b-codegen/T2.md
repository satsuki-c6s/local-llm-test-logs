```html
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>ブロック崩しゲーム</title>
<style>
  body {
    margin: 0;
    display: flex;
    flex-direction: column;
    align-items: center;
    background: #1a1a2e;
    font-family: sans-serif;
    color: #fff;
  }
  h1 { margin: 10px 0; font-size: 20px; }
  canvas {
    background: #16213e;
    border: 2px solid #0f3460;
    border-radius: 6px;
  }
  p { font-size: 13px; color: #aaa; }
</style>
</head>
<body>
<h1>ブロック崩し</h1>
<canvas id="game" width="600" height="500"></canvas>
<p>← → キーでパドルを動かす / ゲームオーバー時は Enter キーでリスタート</p>

<script>
(function () {
  const canvas = document.getElementById('game');
  const ctx = canvas.getContext('2d');
  const W = canvas.width;
  const H = canvas.height;

  // ---- 状態 ----
  let state = 'ready'; // ready | playing | gameover | clear
  let score = 0;

  // ---- パドル ----
  const paddle = {
    w: 90, h: 12,
    x: (W - 90) / 2,
    y: H - 30,
    speed: 7
  };

  // ---- ボール ----
  const ball = {
    r: 8,
    x: W / 2,
    y: H - 50,
    dx: 4,
    dy: -4
  };

  // ---- ブロック ----
  const brick = {
    rows: 5,
    cols: 8,
    w: 60, h: 20,
    padding: 8,
    offsetTop: 50,
    offsetLeft: 0,
    colors: ['#e94560', '#f5a623', '#f7d948', '#4ecdc4', '#5b8def']
  };
  brick.offsetLeft = (W - (brick.cols * (brick.w + brick.padding) - brick.padding)) / 2;

  let bricks = [];

  function initBricks() {
    bricks = [];
    for (let r = 0; r < brick.rows; r++) {
      for (let c = 0; c < brick.cols; c++) {
        bricks.push({
          x: brick.offsetLeft + c * (brick.w + brick.padding),
          y: brick.offsetTop + r * (brick.h + brick.padding),
          color: brick.colors[r],
          alive: true
        });
      }
    }
  }

  function reset() {
    score = 0;
    paddle.x = (W - paddle.w) / 2;
    ball.x = W / 2;
    ball.y = H - 50;
    const angle = (Math.random() * 0.6 - 0.3) * Math.PI; // 上向きに少しばらつき
    const speed = 4.5;
    ball.dx = Math.sin(angle) * speed;
    ball.dy = -Math.cos(angle) * speed;
    initBricks();
    state = 'playing';
  }

  // ---- 入力 ----
  const keys = { left: false, right: false };

  document.addEventListener('keydown', (e) => {
    if (e.key === 'ArrowLeft') { keys.left = true; e.preventDefault(); }
    if (e.key === 'ArrowRight') { keys.right = true; e.preventDefault(); }
    if (e.key === 'Enter' && (state === 'gameover' || state === 'clear')) reset();
  });
  document.addEventListener('keyup', (e) => {
    if (e.key === 'ArrowLeft') keys.left = false;
    if (e.key === 'ArrowRight') keys.right = false;
  });

  // ---- 更新 ----
  function update() {
    if (state !== 'playing') return;

    // パドル移動
    if (keys.left) paddle.x -= paddle.speed;
    if (keys.right) paddle.x += paddle.speed;
    paddle.x = Math.max(0, Math.min(W - paddle.w, paddle.x));

    // ボール移動
    ball.x += ball.dx;
    ball.y += ball.dy;

    // 左右の壁
    if (ball.x - ball.r < 0) {
      ball.x = ball.r;
      ball.dx = Math.abs(ball.dx);
    } else if (ball.x + ball.r > W) {
      ball.x = W - ball.r;
      ball.dx = -Math.abs(ball.dx);
    }

    // 天井
    if (ball.y - ball.r < 0) {
      ball.y = ball.r;
      ball.dy = Math.abs(ball.dy);
    }

    // 床 → ゲームオーバー
    if (ball.y + ball.r > H) {
      state = 'gameover';
      return;
    }

    // パドル当たり
    if (
      ball.dy > 0 &&
      ball.y + ball.r >= paddle.y &&
      ball.y + ball.r <= paddle.y + paddle.h + Math.abs(ball.dy) &&
      ball.x >= paddle.x - ball.r &&
      ball.x <= paddle.x + paddle.w + ball.r
    ) {
      // パドル上の当たり位置で反射角を変える
      const hitPos = (ball.x - (paddle.x + paddle.w / 2)) / (paddle.w / 2); // -1〜1
      const maxAngle = Math.PI / 3; // 最大60度
      const angle = hitPos * maxAngle;
      const speed = Math.hypot(ball.dx, ball.dy);
      ball.dx = Math.sin(angle) * speed;
      ball.dy = -Math.cos(angle) * speed;
      ball.y = paddle.y - ball.r;
    }

    // ブロック当たり
    for (const b of bricks) {
      if (!b.alive) continue;
      if (
        ball.x + ball.r > b.x &&
        ball.x - ball.r < b.x + brick.w &&
        ball.y + ball.r > b.y &&
        ball.y - ball.r < b.y + brick.h
      ) {
        b.alive = false;
        score += 10;

        // 反射方向の判定（どの面から当たったか）
        const overlapX = Math.min(ball.x + ball.r - b.x, b.x + brick.w - (ball.x - ball.r));
        const overlapY = Math.min(ball.y + ball.r - b.y, b.y + brick.h - (ball.y - ball.r));
        if (overlapX < overlapY) {
          ball.dx = -ball.dx;
        } else {
          ball.dy = -ball.dy;
        }
        break; // 1フレームで1ブロックだけ処理
      }
    }

    // クリア判定
    if (bricks.every(b => !b.alive)) {
      state = 'clear';
    }
  }

  // ---- 描画 ----
  function draw() {
    ctx.clearRect(0, 0, W, H);

    // ブロック
    for (const b of bricks) {
      if (!b.alive) continue;
      ctx.fillStyle = b.color;
      ctx.fillRect(b.x, b.y, brick.w, brick.h);
      ctx.strokeStyle = 'rgba(255,255,255,0.25)';
      ctx.strokeRect(b.x + 0.5, b.y + 0.5, brick.w - 1, brick.h - 1);
    }

    // パドル
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(paddle.x, paddle.y, paddle.w, paddle.h);

    // ボール
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
    ctx.fillStyle = '#ffd166';
    ctx.fill();

    // スコア
    ctx.fillStyle = '#fff';
    ctx.font = '16px sans-serif';
    ctx.textAlign = 'left';
    ctx.fillText('スコア: ' + score, 12, 24);

    // メッセージ
    if (state !== 'playing') {
      ctx.fillStyle = 'rgba(0,0,0,0.6)';
      ctx.fillRect(0, 0, W, H);

      ctx.textAlign = 'center';
      ctx.fillStyle = '#fff';
      ctx.font = 'bold 36px sans-serif';

      if (state === 'gameover') {
        ctx.fillText('ゲームオーバー', W / 2, H / 2 - 20);
        ctx.font = '18px sans-serif';
        ctx.fillText('スコア: ' + score, W / 2, H / 2 + 15);
        ctx.fillText('Enter キーでリスタート', W / 2, H / 2 + 45);
      } else if (state === 'clear') {
        ctx.fillStyle = '#4ecdc4';
        ctx.fillText('クリア！おめでとう！', W / 2, H / 2 - 20);
        ctx.fillStyle = '#fff';
        ctx.font = '18px sans-serif';
        ctx.fillText('スコア: ' + score, W / 2, H / 2 + 15);
        ctx.fillText('Enter キーでリスタート', W / 2, H / 2 + 45);
      }
    }
  }

  // ---- メインループ ----
  function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
  }

  initBricks();
  loop();
})();
</script>
</body>
</html>
```

## 機能の説明

| 要件 | 実装内容 |
|---|---|
| (1) canvas描画 | 600×500 の canvas に `requestAnimationFrame` で毎フレーム描画 |
| (2) ボール自動移動・壁天井で跳ね返り | 速度ベクトル `dx/dy` で移動し、左右壁・天井で符号反転 |
| (3) パドル操作 | `ArrowLeft` / `ArrowRight` キーで移動（キーの同時押しにも対応） |
| (4) パドルで打ち返し | パドル上の当たり位置に応じて反射角が変わる（中央で真上、端で斜め） |
| (5) ブロック複数段 | 5段×8列、ボールが当たると消滅（当たり面を判定して正しく反射） |
| (6) ゲームオーバー/クリア表示 | 床に落ちたら「ゲームオーバー」、全消しで「クリア！」表示、Enterでリスタート |
| (7) スコア表示 | ブロック1つにつき10点、画面上部と終了画面に表示 |

ファイルを `.html` で保存してブラウザで開くだけで遊べます。
